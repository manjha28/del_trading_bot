from __future__ import annotations

from dataclasses import dataclass, field
from datetime import datetime, timezone
import importlib
import math
from pathlib import Path

from typing import Any, Literal, Mapping, Protocol, runtime_checkable

import pandas as pd

from .market_data import Instrument, TIMEFRAME_SECONDS


Direction = Literal["LONG", "SHORT"]


@dataclass(frozen=True)
class StrategySpecification:
    strategy_id: str
    version: str
    timeframes: tuple[str, ...]
    required_fields: tuple[str, ...]
    regimes: tuple[str, ...] = ()
    description: str = ""

    def __post_init__(self) -> None:
        if not self.strategy_id or not self.version:
            raise ValueError("strategy_id and version are required")
        unsupported = set(self.timeframes).difference(TIMEFRAME_SECONDS)
        if unsupported:
            raise ValueError(f"unknown strategy timeframes: {sorted(unsupported)}")


@dataclass(frozen=True)
class StrategyContext:
    instrument: Instrument
    as_of: datetime
    frames: Mapping[str, pd.DataFrame]
    regime: Any = None
    profile: Any = None
    provenance: Mapping[str, str] = field(default_factory=dict)
    state: Mapping[str, Any] = field(default_factory=dict)

    def __post_init__(self) -> None:
        if self.as_of.tzinfo is None or self.as_of.utcoffset() is None:
            raise ValueError("strategy context as_of must be timezone-aware")
        object.__setattr__(self, "as_of", self.as_of.astimezone(timezone.utc))

    def completed_frame(self, timeframe: str) -> pd.DataFrame:
        if timeframe not in TIMEFRAME_SECONDS:
            raise ValueError(f"unsupported timeframe: {timeframe}")
        frame = self.frames.get(timeframe)
        if frame is None:
            return pd.DataFrame(columns=["time", "open", "high", "low", "close", "volume"])
        if frame.empty:
            return frame.copy().reset_index(drop=True)
        required = {"time", "open", "high", "low", "close", "volume"}
        missing = required.difference(frame.columns)
        if missing:
            raise ValueError(f"{timeframe} frame missing fields: {sorted(missing)}")
        result = frame.copy()
        parsed_times = pd.to_datetime(result["time"], errors="raise")
        if parsed_times.isna().any() or parsed_times.dt.tz is None:
            raise ValueError(f"{timeframe} timestamps must be valid and timezone-aware")
        result["time"] = parsed_times.dt.tz_convert("UTC")
        if result["time"].duplicated().any():
            raise ValueError(f"{timeframe} frame contains duplicate candle timestamps")
        result = result.sort_values("time")
        close_times = result["time"] + pd.to_timedelta(TIMEFRAME_SECONDS[timeframe], unit="s")
        result = result.loc[close_times <= pd.Timestamp(self.as_of)].reset_index(drop=True)
        return result


@dataclass(frozen=True)
class StrategySignal:
    strategy_id: str
    strategy_version: str
    product_id: str
    symbol: str
    direction: Direction
    decision_time: datetime
    entry_price: float | None
    entry_reference: str
    stop_price: float | None
    stop_logic: str
    target_prices: tuple[float, ...]
    target_logic: str
    invalidation_conditions: tuple[str, ...]
    position_sizing_inputs: Mapping[str, Any]
    provenance: Mapping[str, str]
    diagnostics: Mapping[str, Any] = field(default_factory=dict)

    def __post_init__(self) -> None:
        if self.direction not in ("LONG", "SHORT"):
            raise ValueError("direction must be LONG or SHORT")
        if self.decision_time.tzinfo is None or self.decision_time.utcoffset() is None:
            raise ValueError("signal decision_time must be timezone-aware")
        object.__setattr__(self, "decision_time", self.decision_time.astimezone(timezone.utc))
        if not self.strategy_id or not self.product_id or not self.symbol:
            raise ValueError("signal strategy and instrument identity are required")
        for name in ("entry_price", "stop_price"):
            value = getattr(self, name)
            if value is not None and (not math.isfinite(float(value)) or float(value) <= 0):
                raise ValueError(f"{name} must be a positive finite price")
        targets = tuple(float(value) for value in self.target_prices)
        if any(not math.isfinite(value) or value <= 0 for value in targets):
            raise ValueError("target prices must be positive and finite")
        object.__setattr__(self, "target_prices", targets)


@runtime_checkable
class StrategyAdapter(Protocol):
    specification: StrategySpecification

    def evaluate(self, context: StrategyContext) -> list[StrategySignal]: ...


def _decision_time(value: Any, timeframe: str) -> datetime:
    stamp = pd.Timestamp(value)
    if stamp.tzinfo is None:
        stamp = stamp.tz_localize("UTC")
    return (stamp + pd.Timedelta(seconds=TIMEFRAME_SECONDS[timeframe])).to_pydatetime()


def _mhr_api():
    package_root = Path(__file__).resolve().parents[1] / "strategy-short"
    package_path = str(package_root)
    if package_path not in sys.path:
        sys.path.insert(0, package_path)
    return importlib.import_module("mhr_pump_short")


class MHRExhaustionAdapter:
    specification = StrategySpecification(
        "mhr_exhaustion", "legacy-wrapper-v1", ("1h",),
        ("time", "open", "high", "low", "close", "volume"),
        description="Compatibility wrapper around the existing MHR entry detector.",
    )

    def __init__(self, config: Any = None, entry_paths: set[str] | None = None):
        api = _mhr_api()
        self.api = api
        self.config = config or api.MHRConfig()
        self.entry_paths = frozenset(entry_paths) if entry_paths is not None else None

    def evaluate(self, context: StrategyContext) -> list[StrategySignal]:
        frame = context.completed_frame("1h")
        if frame.empty:
            return []
        prior_price = context.state.get("previous_entry_price")
        is_reentry = bool(context.state.get("is_reentry", False))
        result = self.api.entry_signal(
            frame, len(frame) - 1, self.config,
            is_reentry=is_reentry, previous_entry_price=prior_price,
        )
        if result is None or (
            self.entry_paths is not None and result["entry_path"] not in self.entry_paths
        ):
            return []
        candle = frame.iloc[-1]
        previous_high = float(frame.iloc[-2].high) if len(frame) > 1 else None
        return [StrategySignal(
            strategy_id=self.specification.strategy_id,
            strategy_version=self.specification.version,
            product_id=context.instrument.product_id,
            symbol=context.instrument.symbol,
            direction="SHORT",
            decision_time=_decision_time(candle.time, "1h"),
            entry_price=None,
            entry_reference="next_1h_open",
            stop_price=None,
            stop_logic=(
                "previous completed candle high; legacy ATR fallback when structural "
                "stop cannot produce a valid quantity"
            ),
            target_prices=(),
            target_logic=(
                "legacy MHR minimum reward:risk target, calculated after entry and "
                "independently of this signal"
            ),
            invalidation_conditions=("entry setup ceases to be valid before next-bar entry",),
            position_sizing_inputs={
                "leverage": self.config.leverage,
                "risk_capital_pct": self.config.sl_capital_pct,
                "max_capital_usage_pct": self.config.max_capital_usage_pct,
                "quantity": "deferred to independent risk sizing",
            },
            provenance={"1h": context.provenance.get("1h", "unspecified")},
            diagnostics={**result, "previous_candle_high": previous_high},
        )]


class ScalpPullbackAdapter:
    specification = StrategySpecification(
        "scalp_pullback", "legacy-wrapper-v1", ("5m", "1h"),
        ("time", "open", "high", "low", "close", "volume"),
        regimes=("bullish", "bearish"),
        description="Compatibility wrapper around the existing scalp entry detector.",
    )

    def __init__(self, config: Any = None):
        api = _mhr_api()
        self.api = api
        self.config = config or api.ScalpConfig()

    def evaluate(self, context: StrategyContext) -> list[StrategySignal]:
        frame = context.completed_frame("5m")
        hourly = context.completed_frame("1h")
        minimum = max(self.config.ema_5m_length, self.config.max_pullback_candles + 2)
        if len(frame) <= minimum or len(hourly) < self.config.ema_1h_length:
            return []
        ema = self.api.ema(hourly["close"].astype(float), self.config.ema_1h_length)
        if pd.isna(ema.iloc[-1]):
            return []
        trend = "BULLISH" if float(hourly.close.iloc[-1]) > float(ema.iloc[-1]) else "BEARISH"
        frame["ema20"] = self.api.ema(frame["close"].astype(float), self.config.ema_5m_length)
        frame["atr14"] = self.api.atr(frame, 14)
        result = self.api.scalp_entry(frame, len(frame) - 1, trend, self.config)
        if result is None:
            return []
        candle = frame.iloc[-1]
        direction: Direction = result["direction"]
        return [StrategySignal(
            strategy_id=self.specification.strategy_id,
            strategy_version=self.specification.version,
            product_id=context.instrument.product_id,
            symbol=context.instrument.symbol,
            direction=direction,
            decision_time=_decision_time(candle.time, "5m"),
            entry_price=None,
            entry_reference="next_5m_open",
            stop_price=float(result["sl_price"]),
            stop_logic="legacy pullback extreme plus configured buffer",
            target_prices=(),
            target_logic=(
                f"partial take profit at {self.config.tp1_rr}R then trail remaining "
                f"position with {self.config.trail_lock_rr}R lock"
            ),
            invalidation_conditions=("stop loss hit", "trend filter reverses"),
            position_sizing_inputs={
                "leverage": self.config.leverage,
                "risk_capital_pct": self.config.sl_capital_pct,
                "max_capital_usage_pct": self.config.max_capital_usage_pct,
                "quantity": "deferred to independent risk sizing",
            },
            provenance={
                "5m": context.provenance.get("5m", "unspecified"),
                "1h": context.provenance.get("1h", "unspecified"),
            },
            diagnostics={**result, "trend": trend},
        )]


class FastSlowMAAdapter:
    def __init__(self, config: Any = None, timeframe: str = "1h"):
        if timeframe not in TIMEFRAME_SECONDS:
            raise ValueError(f"unsupported timeframe: {timeframe}")
        self.timeframe = timeframe
        self.long_strategy = importlib.import_module("long_strategy")
        self.config = config or self.long_strategy.FAST_SLOW_DEFAULT_CONFIG
        self.specification = StrategySpecification(
            "fast_slow_ma", "legacy-wrapper-v1", (timeframe,),
            ("time", "open", "high", "low", "close", "volume"),
            description="Compatibility wrapper around Fast/Slow MA signals.",
        )

    def evaluate(self, context: StrategyContext) -> list[StrategySignal]:
        frame = context.completed_frame(self.timeframe)
        if frame.empty:
            return []
        results = []
        for direction, function in (
            ("LONG", self.long_strategy.get_long_signal),
            ("SHORT", self.long_strategy.get_short_signal),
        ):
            raw = function(frame, self.config)
            if raw is None:
                continue
            results.append(StrategySignal(
                strategy_id=self.specification.strategy_id,
                strategy_version=self.specification.version,
                product_id=context.instrument.product_id,
                symbol=context.instrument.symbol,
                direction=direction,
                decision_time=_decision_time(raw["entry_candle_time"], self.timeframe),
                entry_price=float(raw["entry"]),
                entry_reference="legacy signal candle close",
                stop_price=float(raw["stop_loss"]),
                stop_logic="legacy ATR-based stop",
                target_prices=(float(raw["take_profit"]),),
                target_logic="legacy configured risk:reward target",
                invalidation_conditions=("opposite fast/slow moving-average crossover",),
                position_sizing_inputs={
                    "legacy_quantity_reference": raw["quantity"],
                    "leverage": raw["leverage"],
                    "quantity": "recomputed by independent risk sizing",
                },
                provenance={self.timeframe: context.provenance.get(self.timeframe, "unspecified")},
                diagnostics=raw,
            ))
        return results


class VPVRStrategyAdapter:
    def __init__(self, strategy_id: str):
        from vpvr import strategies

        detectors = {
            "poc_reclaim": strategies.poc_reclaim,
            "value_area_rejection": strategies.value_area_rejection,
            "lvn_breakout": strategies.lvn_breakout,
        }
        try:
            self.detector = detectors[strategy_id]
        except KeyError as exc:
            raise ValueError(f"unknown VPVR strategy: {strategy_id}") from exc
        self.specification = StrategySpecification(
            strategy_id, "legacy-wrapper-v1", ("1m", "3m", "15m"),
            ("time", "open", "high", "low", "close", "volume"),
            description="Compatibility wrapper around one VPVR strategy detector.",
        )

    def evaluate(self, context: StrategyContext) -> list[StrategySignal]:
        if context.regime is None or context.profile is None:
            return []
        frames = {tf: context.completed_frame(tf) for tf in self.specification.timeframes}
        result = self.detector(
            context.instrument.symbol, frames["1m"], frames["3m"], frames["15m"],
            context.profile, context.regime,
        )
        if result is None:
            return []
        return [StrategySignal(
            strategy_id=result.strategy,
            strategy_version=self.specification.version,
            product_id=context.instrument.product_id,
            symbol=result.symbol,
            direction=result.direction,
            decision_time=_decision_time(result.timestamp, "1m"),
            entry_price=float(result.entry),
            entry_reference="legacy trigger entry",
            stop_price=float(result.stop_loss),
            stop_logic="legacy strategy stop level",
            target_prices=tuple(float(price) for price in result.targets),
            target_logic="legacy ordered volume-profile targets",
            invalidation_conditions=tuple(result.invalidation),
            position_sizing_inputs={"quantity": "deferred to independent risk sizing"},
            provenance={
                timeframe: context.provenance.get(timeframe, "unspecified")
                for timeframe in self.specification.timeframes
            },
            diagnostics=result.as_dict(),
        )]

def _frame(context: ResearchVariantContext, timeframe: str, minimum: int) -> pd.DataFrame | None:
    frame = context.strategy.completed_frame(timeframe)
    if len(frame) < minimum:
        return None
    return frame


def _signal(
    context: StrategyContext,
    strategy_id: str,
    timeframe: str,
    direction: str,
    entry: float,
    stop: float,
    reasons: tuple[str, ...],
    provenance_extra: Mapping[str, str] | None = None,
) -> StrategySignal | None:
    if not all(math.isfinite(value) and value > 0 for value in (entry, stop)):
        return None
    risk = stop - entry if direction == "SHORT" else entry - stop
    if risk <= 0:
        return None
    target = entry - 2 * risk if direction == "SHORT" else entry + 2 * risk
    candle = context.completed_frame(timeframe).iloc[-1]
    provenance = {timeframe: context.provenance.get(timeframe, "unspecified")}
    provenance.update(provenance_extra or {})
    return StrategySignal(
        strategy_id=strategy_id,
        strategy_version="research-variant-v1",
        product_id=context.instrument.product_id,
        symbol=context.instrument.symbol,
        direction=direction,
        decision_time=_decision_time(candle.time, timeframe),
        entry_price=None,
        entry_reference=f"next_{timeframe}_open",
        stop_price=stop,
        stop_logic="recent completed-candle structure",
        target_prices=(target,),
        target_logic="fixed 2R research target; not optimized or execution-adjusted",
        invalidation_conditions=("structural stop breached",),
        position_sizing_inputs={"quantity": "deferred to independent risk sizing"},
        provenance=provenance,
        diagnostics={"reasons": reasons, "research_only": True},
    )


def _parabolic_failure(ctx: ResearchVariantContext) -> list[StrategySignal]:
    frame = _frame(ctx, "1h", 48)
    if frame is None:
        return []
    closes = frame.close.astype(float)
    returns = closes.pct_change()
    current_return = closes.iloc[-1] / closes.iloc[-25] - 1
    baseline = returns.iloc[-48:-24].dropna().abs()
    robust_scale = float(baseline.median())
    range_pct = (frame.high.astype(float) - frame.low.astype(float)) / closes
    prior_range = range_pct.iloc[-48:-1].dropna()
    if robust_scale <= 0 or prior_range.empty:
        return []
    displacement = float(current_return)
    # Both event displacement and current range are compared with this symbol's own history.
    if displacement < max(4 * robust_scale, float(baseline.quantile(.9)) * 8):
        return []
    local_lows = frame.low.astype(float).iloc[-5:-1]
    close = float(closes.iloc[-1])
    if close >= float(local_lows.min()):
        return []
    stop = float(frame.high.astype(float).iloc[-3:].max())
    result = _signal(
        ctx.strategy, "parabolic_auction_failure_short", "1h", "SHORT",
        close, stop,
        ("per-symbol 24h displacement exceeded its rolling robust baseline",
         "completed candle closed below prior four-candle support",
         f"latest range percentile={float((prior_range <= range_pct.iloc[-1]).mean()):.3f}"),
        {"event_anchor": "rolling OHLCV high; order-book auction data unavailable"},
    )
    return [result] if result else []


def _failed_high_vwap_retest(ctx: ResearchVariantContext) -> list[StrategySignal]:
    frame = _frame(ctx, "1h", 20)
    if frame is None:
        return []
    recent = frame.iloc[-8:].copy()
    high_idx = recent.high.astype(float).idxmax()
    anchor_position = int(frame.index.get_loc(high_idx))
    after_high = frame.iloc[anchor_position + 1:]
    if len(after_high) < 2:
        return []
    anchor = float(frame.loc[high_idx, "high"])
    volume = frame.volume.astype(float).clip(lower=0)
    typical = (frame.high.astype(float) + frame.low.astype(float) + frame.close.astype(float)) / 3
    weights = volume.iloc[anchor_position:]
    total_volume = float(weights.sum())
    if total_volume <= 0:
        return []
    anchored_vwap = float((typical.iloc[anchor_position:] * weights).sum() / total_volume)
    previous, current = after_high.iloc[-2], after_high.iloc[-1]
    close = float(current.close)
    retest = float(current.high) >= anchor * .995 and close < anchor
    failed_vwap = close < anchored_vwap
    lower_high = float(current.high) < anchor
    if not (retest and failed_vwap and lower_high):
        return []
    stop = max(float(current.high), float(previous.high), anchor) * 1.001
    result = _signal(
        ctx.strategy, "failed_high_vwap_retest", "1h", "SHORT", close, stop,
        ("retest rejected below the recent completed-candle high",
         "close below OHLCV-anchored VWAP from swing-high candle",
         "lower-high structure confirmed"),
        {"vwap": "candle typical-price volume-weighted, anchored to detected high"},
    )
    return [result] if result else []


def _liquidity_sweep_reclaim(ctx: ResearchVariantContext) -> list[StrategySignal]:
    frame = _frame(ctx, "5m", 22)
    if frame is None:
        return []
    prior = frame.iloc[-21:-1]
    current = frame.iloc[-1]
    support = float(prior.low.min())
    resistance = float(prior.high.max())
    close, high, low = float(current.close), float(current.high), float(current.low)
    if low < support and close > support:
        direction, entry, stop = "LONG", close, low * .999
    elif high > resistance and close < resistance:
        direction, entry, stop = "SHORT", close, high * 1.001
    else:
        return []
    signal = _signal(
        ctx.strategy, "liquidity_sweep_reclaim", "5m", direction, entry, stop,
        ("completed candle breached a prior 20-candle extreme and reclaimed it",
         "price-structure proxy only; no resting liquidity inference"),
    )
    return [signal] if signal else []


def _trend_continuation(ctx: ResearchVariantContext) -> list[StrategySignal]:
    frame = _frame(ctx, "4h", 26)
    if frame is None:
        return []
    close = frame.close.astype(float)
    baseline = close.iloc[-25:-1]
    mean = float(baseline.mean())
    std = float(baseline.std(ddof=0))
    current = float(close.iloc[-1])
    prior_high = float(frame.high.astype(float).iloc[-25:-1].max())
    prior_low = float(frame.low.astype(float).iloc[-25:-1].min())
    if current > mean and current > prior_high:
        stop = float(frame.low.astype(float).iloc[-4:-1].min())
        direction = "LONG"
    elif current < mean and current < prior_low:
        stop = float(frame.high.astype(float).iloc[-4:-1].max())
        direction = "SHORT"
    else:
        return []
    if std <= 0:
        return []
    signal = _signal(
        ctx.strategy, "multi_day_trend_continuation", "4h",
        direction, current, stop,
        ("4h close broke the previous six-day range in the direction of its rolling mean",
         "trend condition calculated using completed candles only"),
    )
    return [signal] if signal else []


def _volatility_breakout(ctx: ResearchVariantContext) -> list[StrategySignal]:
    frame = _frame(ctx, "1h", 48)
    if frame is None:
        return []
    ranges = (frame.high.astype(float) - frame.low.astype(float)) / frame.close.astype(float)
    history = ranges.iloc[-48:-1]
    current_range = float(ranges.iloc[-1])
    if history.empty or current_range < float(history.quantile(.9)):
        return []
    volume = frame.volume.astype(float)
    typical_volume = float(volume.iloc[-48:-1].median())
    if typical_volume <= 0 or float(volume.iloc[-1]) < typical_volume:
        return []
    prior = frame.iloc[-25:-1]
    close = float(frame.close.iloc[-1])
    if close > float(prior.high.max()):
        direction = "LONG"
        stop = float(frame.low.iloc[-1])
    elif close < float(prior.low.min()):
        direction = "SHORT"
        stop = float(frame.high.iloc[-1])
    else:
        return []
    signal = _signal(
        ctx.strategy, "volatility_expansion_breakout", "1h",
        direction, close, stop,
        ("1h range exceeded this symbol's preceding 47-hour 90th percentile",
         "volume exceeded its preceding 47-hour median",
         "close broke the preceding 24-hour completed-candle range"),
    )
    return [signal] if signal else []


def _trade_flow_momentum(ctx: ResearchVariantContext) -> list[StrategySignal]:
    if not ctx.trade_history_complete:
        return []
    frame = _frame(ctx, "1m", 15)
    if frame is None:
        return []
    as_of = ctx.strategy.as_of
    trades = [
        trade for trade in ctx.trades
        if trade.timestamp <= as_of and trade.timestamp >= as_of - pd.Timedelta(minutes=5)
    ]
    if len(trades) < 20 or any(trade.aggressor_side == "unknown" for trade in trades):
        return []
    buy = sum(trade.quantity for trade in trades if trade.aggressor_side == "buy")
    sell = sum(trade.quantity for trade in trades if trade.aggressor_side == "sell")
    total = buy + sell
    if total <= 0 or abs(buy - sell) / total < .25:
        return []
    recent = frame.iloc[-6:]
    direction = "LONG" if buy > sell else "SHORT"
    close = float(recent.close.iloc[-1])
    price_change = close / float(recent.close.iloc[0]) - 1
    if (direction == "LONG" and price_change <= 0) or (direction == "SHORT" and price_change >= 0):
        return []
    stop = float(recent.low.min()) if direction == "LONG" else float(recent.high.max())
    signal = _signal(
        ctx.strategy, "trade_flow_momentum", "1m", direction, close, stop,
        ("complete recent trade tape has explicit aggressor-side labels",
         f"five-minute signed quantity imbalance={((buy-sell)/total):.3f}",
         "price progress is aligned with signed trade flow"),
        {"trades": "exchange-provided aggressor-side labels; unknown-side events rejected"},
    )
    return [signal] if signal else []


def _book_imbalance(ctx: ResearchVariantContext) -> list[StrategySignal]:
    frame = _frame(ctx, "1m", 15)
    if frame is None:
        return []
    snapshots = sorted(
        (book for book in ctx.order_books
         if book.observed_at <= ctx.strategy.as_of
         and book.integrity == "synchronized"
         and ctx.strategy.as_of - book.observed_at <= pd.Timedelta(minutes=2)),
        key=lambda book: book.observed_at,
    )
    if len(snapshots) < 2:
        return []
    first, last = snapshots[-2:]
    if last.observed_at - first.observed_at <= pd.Timedelta(0):
        return []
    depth = min(5, len(last.bids), len(last.asks))
    if depth < 1:
        return []
    bid_qty = sum(level.quantity for level in last.bids[:depth])
    ask_qty = sum(level.quantity for level in last.asks[:depth])
    total = bid_qty + ask_qty
    if total <= 0:
        return []
    imbalance = (bid_qty - ask_qty) / total
    if abs(imbalance) < .3:
        return []
    direction = "LONG" if imbalance > 0 else "SHORT"
    close = float(frame.close.iloc[-1])
    spread = last.asks[0].price - last.bids[0].price
    microprice = (
        last.asks[0].price * last.bids[0].quantity
        + last.bids[0].price * last.asks[0].quantity
    ) / max(last.bids[0].quantity + last.asks[0].quantity, 1e-12)
    if (direction == "LONG" and microprice <= close) or (direction == "SHORT" and microprice >= close):
        return []
    stop = close - 2 * spread if direction == "LONG" else close + 2 * spread
    signal = _signal(
        ctx.strategy, "order_book_imbalance_microprice", "1m",
        direction, close, stop,
        ("two fresh synchronized order-book snapshots available",
         f"top-{depth} depth imbalance={imbalance:.3f}",
         "microprice is aligned with imbalance"),
        {"book": "snapshot data only; no depletion/replenishment inference"},
    )
    return [signal] if signal else []

    "liquidity_sweep_reclaim": _liquidity_sweep_reclaim,
    "multi_day_trend_continuation": _trend_continuation,
    "volatility_expansion_breakout": _volatility_breakout,
    "trade_flow_momentum": _trade_flow_momentum,
    "order_book_imbalance_microprice": _book_imbalance,
}


class ResearchVariantSuite:
    @property
    def catalog(self) -> tuple[ResearchVariant, ...]:
        return VARIANT_CATALOG

    def evaluate(
        self, variant_id: str, context: ResearchVariantContext
    ) -> list[StrategySignal]:
        if variant_id not in _BY_ID:
            raise ValueError(f"unknown research variant: {variant_id}")
        variant = _BY_ID[variant_id]
        evaluator = _EVALUATORS.get(variant_id)
        if evaluator is None:
            return []
        return evaluator(context)

    def evaluate_available(
        self, context: ResearchVariantContext
    ) -> dict[str, list[StrategySignal]]:
        return {
            variant.strategy_id: self.evaluate(variant.strategy_id, context)
            for variant in VARIANT_CATALOG
            if variant.strategy_id in _EVALUATORS
        }

"""Three VPVR strategy detectors with completed-candle sequencing and target hierarchy."""
from __future__ import annotations
from dataclasses import dataclass, asdict
import pandas as pd
from .regime import MarketRegime
from .indicators import ema, session_vwap, volume_expansion

@dataclass(frozen=True)
class StrategySignal:
    symbol: str
    strategy: str
    direction: str
    entry: float
    stop_loss: float
    targets: tuple[float, ...]
    risk_reward: float
    timestamp: str
    reasons: tuple[str, ...]
    invalidation: tuple[str, ...]
    setup_level: float
    expiry_candles: int = 5
    def as_dict(self): return asdict(self)

def _valid(frame, minimum=10):
    required={"time","open","high","low","close","volume"}
    return frame is not None and len(frame)>=minimum and required.issubset(frame.columns) and pd.to_datetime(frame.time,utc=True).is_monotonic_increasing

def _after_3m_confirmation(c1,c3):
    three_min_close=pd.Timestamp(c3.time.iloc[-1])+pd.Timedelta(minutes=3)
    return pd.Timestamp(c1.time.iloc[-1])>=three_min_close

def _rr(entry,stop,target,direction):
    risk=entry-stop if direction=="LONG" else stop-entry
    reward=target-entry if direction=="LONG" else entry-target
    return reward/risk if risk>0 else 0.0

def _make(symbol,name,direction,entry,stop,targets,timestamp,reasons,invalidation,level,min_rr=2.0):
    ordered=sorted({float(x) for x in targets if (float(x)>entry if direction=="LONG" else float(x)<entry)},reverse=(direction=="SHORT"))[:3]
    if not ordered:return None
    rr=_rr(entry,stop,ordered[0],direction)
    if rr<min_rr:return None
    return StrategySignal(symbol,name,direction,float(entry),float(stop),tuple(ordered),float(rr),pd.Timestamp(timestamp).isoformat(),tuple(reasons),tuple(invalidation),float(level))

def _micro_context(c1):
    e=ema(c1.close,20);v=session_vwap(c1)
    if pd.isna(e.iloc[-1]) or pd.isna(v.iloc[-1]):return None,None
    return float(e.iloc[-1]),float(v.iloc[-1])

def poc_reclaim(symbol,c1,c3,c15,profile,regime,min_rr=2.0,volume_multiple=1.0):
    if not all(_valid(x) for x in (c1,c3,c15)) or not _after_3m_confirmation(c1,c3):return None
    previous,confirmation,retest=c3.iloc[-2],c3.iloc[-1],c1.iloc[-1]
    expansion=volume_expansion(c3)
    if expansion is None or expansion<volume_multiple:return None
    if regime.regime not in (MarketRegime.BULLISH_TREND,MarketRegime.BEARISH_TREND,MarketRegime.RANGE):return None
    e20,vwap=_micro_context(c1)
    if e20 is None:return None
    poc=float(profile.poc)
    if float(previous.close)<=poc<float(confirmation.close) and float(retest.low)<=poc*1.001 and float(retest.close)>poc and float(retest.close)>float(retest.open) and float(retest.close)>=e20 and float(retest.close)>=vwap and regime.regime!=MarketRegime.BEARISH_TREND:
        entry=float(retest.high);stop=min(float(retest.low),float(c1.low.tail(4).min()))*(1-0.001)
        targets=[x for x in (*profile.hvns,profile.vah) if x>entry]
        return _make(symbol,"poc_reclaim","LONG",entry,stop,targets,retest.time,("completed 3m close reclaimed POC","3m volume expanded vs previous five completed bars","subsequent 1m bullish POC retest","1m close above EMA20 and session VWAP"),("1m close below POC","15m regime turns bearish"),poc,min_rr)
    if float(previous.close)>=poc>float(confirmation.close) and float(retest.high)>=poc*.999 and float(retest.close)<poc and float(retest.close)<float(retest.open) and float(retest.close)<=e20 and float(retest.close)<=vwap and regime.regime!=MarketRegime.BULLISH_TREND:
        entry=float(retest.low);stop=max(float(retest.high),float(c1.high.tail(4).max()))*(1+0.001)
        targets=[x for x in (*profile.hvns,profile.val) if x<entry]
        return _make(symbol,"poc_reclaim","SHORT",entry,stop,targets,retest.time,("completed 3m close lost POC","3m volume expanded vs previous five completed bars","subsequent 1m bearish POC retest","1m close below EMA20 and session VWAP"),("1m close above POC","15m regime turns bullish"),poc,min_rr)
    return None

def value_area_rejection(symbol,c1,c3,c15,profile,regime,min_rr=2.0,volume_multiple=1.0):
    if not all(_valid(x) for x in (c1,c3,c15)) or regime.regime!=MarketRegime.RANGE:return None
    candle=c1.iloc[-1];expansion=volume_expansion(c1)
    if expansion is None or expansion<volume_multiple:return None
    e20,vwap=_micro_context(c1)
    if e20 is None:return None
    body=abs(float(candle.close)-float(candle.open));lower=float(min(candle.open,candle.close))-float(candle.low);upper=float(candle.high)-float(max(candle.open,candle.close))
    near_vwap=abs(float(candle.close)-vwap)/max(vwap,1e-12)<=0.003
    if float(candle.low)<=profile.val*1.001 and float(candle.close)>profile.val and float(candle.close)>float(candle.open) and lower>body and (float(candle.close)>=vwap or near_vwap) and float(candle.close)>=e20:
        return _make(symbol,"value_area_rejection","LONG",float(candle.high),float(candle.low)*(1-0.001),[profile.poc,profile.vah],candle.time,("range regime","VAL rejection wick","1m volume expansion","EMA20/VWAP context non-adverse"),("1m close below VAL","15m value area expands lower"),profile.val,min_rr)
    if float(candle.high)>=profile.vah*.999 and float(candle.close)<profile.vah and float(candle.close)<float(candle.open) and upper>body and (float(candle.close)<=vwap or near_vwap) and float(candle.close)<=e20:
        return _make(symbol,"value_area_rejection","SHORT",float(candle.low),float(candle.high)*(1+0.001),[profile.poc,profile.val],candle.time,("range regime","VAH rejection wick","1m volume expansion","EMA20/VWAP context non-adverse"),("1m close above VAH","15m value area expands higher"),profile.vah,min_rr)
    return None

def lvn_breakout(symbol,c1,c3,c15,profile,regime,min_rr=2.0,volume_multiple=1.0):
    if not all(_valid(x) for x in (c1,c3,c15)) or not _after_3m_confirmation(c1,c3):return None
    candle=c1.iloc[-1];previous,confirmation=c3.iloc[-2],c3.iloc[-1];expansion=volume_expansion(c3)
    if expansion is None or expansion<volume_multiple or regime.regime not in (MarketRegime.BULLISH_TREND,MarketRegime.BEARISH_TREND,MarketRegime.HIGH_VOL):return None
    e20,vwap=_micro_context(c1)
    if e20 is None:return None
    for boundary in profile.lvns:
        level=float(boundary)
        if float(previous.close)<=level<float(confirmation.close) and float(candle.low)<=level*1.001 and float(candle.close)>level and float(candle.close)>=e20 and float(candle.close)>=vwap and regime.regime!=MarketRegime.BEARISH_TREND:
            targets=[x for x in (*profile.hvns,profile.vah) if x>float(candle.high)]
            return _make(symbol,"lvn_breakout","LONG",float(candle.high),float(candle.low)*(1-0.001),targets,candle.time,("completed 3m close above LVN","3m volume expansion","subsequent 1m retest acceptance","price above EMA20/session VWAP"),("1m close below LVN","15m bias reverses"),level,min_rr)
        if float(previous.close)>=level>float(confirmation.close) and float(candle.high)>=level*.999 and float(candle.close)<level and float(candle.close)<=e20 and float(candle.close)<=vwap and regime.regime!=MarketRegime.BULLISH_TREND:
            targets=[x for x in (*profile.hvns,profile.val) if x<float(candle.low)]
            return _make(symbol,"lvn_breakout","SHORT",float(candle.low),float(candle.high)*(1+0.001),targets,candle.time,("completed 3m close below LVN","3m volume expansion","subsequent 1m retest rejection","price below EMA20/session VWAP"),("1m close above LVN","15m bias reverses"),level,min_rr)
    return None

def generate_signals(symbol,c1,c3,c15,profile,regime,min_rr=2.0,volume_multiple=1.0):
    candidates=[poc_reclaim(symbol,c1,c3,c15,profile,regime,min_rr,volume_multiple),value_area_rejection(symbol,c1,c3,c15,profile,regime,min_rr,volume_multiple),lvn_breakout(symbol,c1,c3,c15,profile,regime,min_rr,volume_multiple)]
    return [signal for signal in candidates if signal is not None]

"""Deterministic Volume Profile with explicit source provenance.

Trade prints are accumulated at their actual execution prices. Candle input is
an approximation: each candle's reported volume is distributed in proportion
to the overlap of its high-low range with each price bin. This is not exact
volume-at-price and must remain labeled as an approximation.
"""
from __future__ import annotations
from dataclasses import dataclass
import numpy as np
import pandas as pd

@dataclass(frozen=True)
class VolumeProfile:
    prices: tuple[float,...]; volumes: tuple[float,...]; poc: float; vah: float; val: float
    hvns: tuple[float,...]; lvns: tuple[float,...]; total_volume: float; source: str
    start: pd.Timestamp|None=None; end: pd.Timestamp|None=None
    def as_dict(self):
        return {
            "poc": self.poc, "vah": self.vah, "val": self.val,
            "hvns": list(self.hvns), "lvns": list(self.lvns),
            "total_volume": self.total_volume, "source": self.source,
            "rows": len(self.prices),
            "price_levels": list(self.prices),
            "volume_at_price": list(self.volumes),
            "start": str(self.start) if self.start is not None else None,
            "end": str(self.end) if self.end is not None else None,
        }

def _value_area(volumes,prices,fraction):
    if len(volumes)==0 or volumes.sum()<=0: raise ValueError("profile has no positive volume")
    poc_idx=int(np.argmax(volumes)); lo=hi=poc_idx; accumulated=float(volumes[poc_idx]); target=float(volumes.sum()*fraction)
    while accumulated<target and (lo>0 or hi<len(volumes)-1):
        left=float(volumes[lo-1]) if lo>0 else -1.0; right=float(volumes[hi+1]) if hi<len(volumes)-1 else -1.0
        if right>left: hi+=1; accumulated+=float(volumes[hi])
        elif lo>0: lo-=1; accumulated+=float(volumes[lo])
        else: break
    return float(prices[poc_idx]),float(prices[hi]),float(prices[lo])

def _nodes(volumes,prices,window,threshold):
    smooth=pd.Series(volumes).rolling(window,center=True,min_periods=1).mean().to_numpy(); maxv=float(smooth.max()) if len(smooth) else 0.0
    hvn=[]; lvn=[]
    for i in range(1,len(smooth)-1):
        if smooth[i]>=smooth[i-1] and smooth[i]>smooth[i+1] and smooth[i]>=maxv*threshold: hvn.append(float(prices[i]))
        if smooth[i]<=smooth[i-1] and smooth[i]<smooth[i+1] and smooth[i]<=maxv*threshold: lvn.append(float(prices[i]))
    return tuple(hvn),tuple(lvn)

def build_volume_profile(candles=None,*,trades=None,rows=60,value_area_pct=.70,smoothing_window=3,node_threshold_pct=.15,start=None,end=None):
    if not 50<=rows<=70: raise ValueError("rows must be between 50 and 70")
    if not .5<=value_area_pct<=.9: raise ValueError("invalid value area fraction")
    if smoothing_window<1 or not 0<node_threshold_pct<1: raise ValueError("invalid node parameters")
    source="trade_prints"
    if trades is not None and not trades.empty:
        if not {"price","volume"}.issubset(trades.columns): raise ValueError("trades require price and volume")
        frame=trades[["price","volume"]].copy()
        frame["price"]=pd.to_numeric(frame.price,errors="coerce"); frame["volume"]=pd.to_numeric(frame.volume,errors="coerce")
        frame=frame.replace([np.inf,-np.inf],np.nan).dropna(); frame=frame[(frame.price>0)&(frame.volume>0)]
        if frame.empty: raise ValueError("no valid trade prints")
        low,high=float(frame.price.min()),float(frame.price.max())
    else:
        if candles is None or candles.empty: raise ValueError("candles or trade prints required")
        required={"high","low","volume"}
        if not required.issubset(candles.columns): raise ValueError(f"candles missing {required-set(candles.columns)}")
        frame=candles[["high","low","volume"]].copy()
        for col in frame.columns: frame[col]=pd.to_numeric(frame[col],errors="coerce")
        frame=frame.replace([np.inf,-np.inf],np.nan).dropna(); frame=frame[(frame.high>0)&(frame.low>0)&(frame.high>=frame.low)&(frame.volume>0)]
        if frame.empty: raise ValueError("no valid OHLCV volume")
        low,high=float(frame.low.min()),float(frame.high.max()); source="ohlcv_uniform_range_approximation"
    if high<=low: high=low+max(abs(low)*1e-6,1e-8)
    edges=np.linspace(low,high,rows+1); prices=(edges[:-1]+edges[1:])/2; volumes=np.zeros(rows,dtype=float)
    if source=="trade_prints":
        idx=np.clip(np.searchsorted(edges,frame.price.to_numpy(),side="right")-1,0,rows-1); np.add.at(volumes,idx,frame.volume.to_numpy())
    else:
        for row in frame.itertuples(index=False):
            lo=float(row.low); hi=float(row.high); vol=float(row.volume)
            if hi==lo:
                idx=int(np.clip(np.searchsorted(edges,lo,side="right")-1,0,rows-1)); volumes[idx]+=vol; continue
            overlap=np.maximum(0.0,np.minimum(edges[1:],hi)-np.maximum(edges[:-1],lo)); total=float(overlap.sum())
            if total>0: volumes+=vol*overlap/total
    poc,vah,val=_value_area(volumes,prices,value_area_pct); hvn,lvn=_nodes(volumes,prices,smoothing_window,node_threshold_pct)
    s=pd.Timestamp(start) if start is not None else (pd.to_datetime(candles.time.min(),utc=True) if candles is not None and "time" in candles else None)
    e=pd.Timestamp(end) if end is not None else (pd.to_datetime(candles.time.max(),utc=True) if candles is not None and "time" in candles else None)
    return VolumeProfile(tuple(map(float,prices)),tuple(map(float,volumes)),poc,vah,val,hvn,lvn,float(volumes.sum()),source,s,e)

def build_fixed_range_profile(candles,**kwargs): return build_volume_profile(candles,**kwargs)
def developing_profile(candles,**kwargs): return build_volume_profile(candles,**kwargs)
def rolling_profiles(candles,rows=60,value_area_pct=.70,window=240,step=1,**kwargs):
    if window<1 or step<1: raise ValueError("window and step must be positive")
    out=[]
    for end_idx in range(window,len(candles)+1,step):
        part=candles.iloc[end_idx-window:end_idx]; out.append(build_volume_profile(part,rows=rows,value_area_pct=value_area_pct,start=part.iloc[0].get("time"),end=part.iloc[-1].get("time"),**kwargs))
    return out
def session_profiles(candles,rows=60,value_area_pct=.70,**kwargs):
    if "time" not in candles: raise ValueError("session profiles require time")
    times=pd.to_datetime(candles.time,utc=True); key=times.dt.strftime("%Y-%m-%d")
    return {day:build_volume_profile(candles.loc[key==day],rows=rows,value_area_pct=value_area_pct,**kwargs) for day in key.drop_duplicates()}

"""Conservative regime classification using multi-timeframe structure, VWAP and profile evolution."""
from __future__ import annotations
from dataclasses import dataclass
from enum import Enum
import pandas as pd
from .indicators import ema, session_vwap

class MarketRegime(str, Enum):
    BULLISH_TREND = "bullish_trending"
    BEARISH_TREND = "bearish_trending"
    RANGE = "range_bound"
    HIGH_VOL = "high_volatility"
    LOW_VOL = "low_volatility"
    UNCERTAIN = "uncertain_no_trade"

@dataclass(frozen=True)
class RegimeResult:
    regime: MarketRegime
    confidence: float
    reasons: tuple[str, ...]


def classify_regime(c15: pd.DataFrame, c3: pd.DataFrame, profile=None, profile_history=None) -> RegimeResult:
    if len(c15) < 30 or len(c3) < 30:
        return RegimeResult(MarketRegime.UNCERTAIN, 0.0, ("insufficient completed higher-timeframe candles",))
    e15 = ema(c15.close, 20)
    e3 = ema(c3.close, 20)
    if pd.isna(e15.iloc[-1]) or pd.isna(e3.iloc[-1]) or pd.isna(e15.iloc[-4]):
        return RegimeResult(MarketRegime.UNCERTAIN, 0.0, ("EMA warmup incomplete",))
    last15, last3 = c15.iloc[-1], c3.iloc[-1]
    atr_pct = float((c15.high - c15.low).tail(14).mean() / max(float(last15.close), 1e-12))
    if atr_pct >= 0.035:
        return RegimeResult(MarketRegime.HIGH_VOL, min(1.0, atr_pct / 0.05), ("15m realized range volatility elevated",))
    if atr_pct <= 0.0015:
        return RegimeResult(MarketRegime.LOW_VOL, 0.8, ("15m realized range volatility compressed",))
    vwap = float(session_vwap(c15).iloc[-1])
    poc_rising = poc_falling = value_expanding = False
    if profile_history and len(profile_history) >= 2:
        first, latest = profile_history[0], profile_history[-1]
        poc_rising = latest.poc > first.poc * 1.001
        poc_falling = latest.poc < first.poc * 0.999
        first_width = first.vah - first.val
        latest_width = latest.vah - latest.val
        value_expanding = first_width > 0 and latest_width > first_width * 1.1
    bull = float(last15.close) > float(e15.iloc[-1]) and float(last3.close) > float(e3.iloc[-1]) and float(e15.iloc[-1]) > float(e15.iloc[-4]) and float(last15.close) >= vwap
    bear = float(last15.close) < float(e15.iloc[-1]) and float(last3.close) < float(e3.iloc[-1]) and float(e15.iloc[-1]) < float(e15.iloc[-4]) and float(last15.close) <= vwap
    if bull and (poc_rising or profile_history is None or len(profile_history) < 2):
        return RegimeResult(MarketRegime.BULLISH_TREND, 0.75, ("15m/3m EMA20 bullish structure", "15m close above session VWAP", "POC rising or unavailable"))
    if bear and (poc_falling or profile_history is None or len(profile_history) < 2):
        return RegimeResult(MarketRegime.BEARISH_TREND, 0.75, ("15m/3m EMA20 bearish structure", "15m close below session VWAP", "POC falling or unavailable"))
    if profile is not None and float(profile.val) <= float(last15.close) <= float(profile.vah) and not value_expanding:
        return RegimeResult(MarketRegime.RANGE, 0.65, ("price inside value area", "value area not expanding materially"))
    if abs(float(last15.close) - vwap) / max(vwap, 1e-12) < 0.002 and not value_expanding:
        return RegimeResult(MarketRegime.RANGE, 0.5, ("price rotating near session VWAP",))
    return RegimeResult(MarketRegime.UNCERTAIN, 0.25, ("timeframes, VWAP or developing profile are not aligned",))

"""VPVR indicator engine. Aggressor delta is calculated only from explicit trade side."""
from __future__ import annotations
import pandas as pd


def ema(series: pd.Series, length: int = 20) -> pd.Series:
    if length < 1:
        raise ValueError("EMA length must be positive")
    return series.ewm(span=length, adjust=False, min_periods=length).mean()


def session_vwap(candles: pd.DataFrame) -> pd.Series:
    """UTC calendar-day VWAP approximation from OHLCV typical price."""
    required={"time","high","low","close","volume"}
    if not required.issubset(candles):
        raise ValueError(f"VWAP missing columns: {required-set(candles.columns)}")
    result=pd.Series(index=candles.index,dtype=float)
    times=pd.to_datetime(candles.time,utc=True)
    sessions=times.dt.strftime("%Y-%m-%d")
    typical=(candles.high+candles.low+candles.close)/3.0
    price_volume=typical*candles.volume
    for session in sessions.drop_duplicates():
        mask=sessions==session
        denominator=candles.loc[mask,"volume"].cumsum().replace(0,float("nan"))
        result.loc[mask]=price_volume.loc[mask].cumsum()/denominator
    return result


def volume_expansion(candles: pd.DataFrame, lookback: int = 5) -> float | None:
    """Latest completed bar volume divided by the preceding N completed bars' average."""
    if lookback<1 or "volume" not in candles or len(candles)<lookback+1:
        return None
    previous=pd.to_numeric(candles.volume.iloc[-lookback-1:-1],errors="coerce").mean()
    current=float(candles.volume.iloc[-1])
    if pd.isna(previous) or previous<=0:
        return None
    return current/float(previous)


def cumulative_volume_delta(trades: pd.DataFrame | None) -> dict:
    """Return signed aggressor volume from REST trade side labels; never infer from candle colour."""
    if trades is None or trades.empty or not {"volume","side"}.issubset(trades.columns):
        return {"available":False,"value":None,"reason":"Aggressor-side trade prints unavailable"}
    side=trades.side.astype(str).str.lower()
    volume=pd.to_numeric(trades.volume,errors="coerce").fillna(0)
    if not side.isin(["buy","sell"]).all():
        return {"available":False,"value":None,"reason":"Trade side values do not identify aggressor buy/sell"}
    return {"available":True,"value":float(volume[side=="buy"].sum()-volume[side=="sell"].sum()),"reason":None}


def trade_delta_bars(trades: pd.DataFrame | None, timeframe: str = "1min") -> pd.DataFrame:
    """Aggregate aggressor buys/sells into bars and compute CVD; missing tape returns empty."""
    if trades is None or trades.empty or not {"time","volume","side"}.issubset(trades.columns):
        return pd.DataFrame(columns=["buy_volume","sell_volume","delta","cvd"])
    side=trades.side.astype(str).str.lower()
    if not side.isin(["buy","sell"]).all():
        return pd.DataFrame(columns=["buy_volume","sell_volume","delta","cvd"])
    frame=trades.copy()
    frame["time"]=pd.to_datetime(frame.time,utc=True)
    frame["volume"]=pd.to_numeric(frame.volume,errors="coerce")
    frame=frame.dropna(subset=["time","volume"])
    frame=frame[frame.volume>0].copy()
    frame["buy_volume"]=frame.volume.where(frame.side.astype(str).str.lower()=="buy",0.0)
    frame["sell_volume"]=frame.volume.where(frame.side.astype(str).str.lower()=="sell",0.0)
    grouped=frame.set_index("time").resample(timeframe).agg(
        buy_volume=("buy_volume","sum"),sell_volume=("sell_volume","sum"))
    grouped["delta"]=grouped.buy_volume-grouped.sell_volume
    grouped["cvd"]=grouped.delta.cumsum()
    return grouped.dropna(how="all")


def delta_divergence(candles: pd.DataFrame, delta_bars: pd.DataFrame, lookback: int = 5) -> dict:
    """Compare aligned price swing extremes with genuine aggressor CVD swing extremes."""
    if lookback < 4 or candles is None or delta_bars is None or len(candles) < lookback:
        return {"available": False, "bullish": False, "bearish": False,
                "reason": "insufficient aligned price and trade-delta history"}
    if "time" not in candles or "cvd" not in delta_bars:
        return {"available": False, "bullish": False, "bearish": False,
                "reason": "timestamped CVD series unavailable"}
    price_frame = candles.tail(lookback).copy()
    price_frame["time"] = pd.to_datetime(price_frame.time, utc=True).dt.floor("min")
    cvd_frame = delta_bars.copy()
    cvd_frame.index = pd.to_datetime(cvd_frame.index, utc=True).floor("min")
    joined = price_frame.set_index("time").join(cvd_frame[["cvd"]], how="left")
    if joined.cvd.isna().mean() > 0.20:
        return {"available": False, "bullish": False, "bearish": False,
                "reason": "trade tape does not cover enough matching 1m bars"}
    price = pd.to_numeric(joined.close, errors="coerce").reset_index(drop=True)
    cvd = pd.to_numeric(joined.cvd, errors="coerce").reset_index(drop=True)
    if price.isna().any() or cvd.isna().any():
        return {"available": False, "bullish": False, "bearish": False,
                "reason": "price/CVD contains missing values"}
    mid = max(2, lookback // 2)
    price_low_1, price_low_2 = float(price.iloc[:mid].min()), float(price.iloc[mid:].min())
    cvd_low_1, cvd_low_2 = float(cvd.iloc[:mid].min()), float(cvd.iloc[mid:].min())
    price_high_1, price_high_2 = float(price.iloc[:mid].max()), float(price.iloc[mid:].max())
    cvd_high_1, cvd_high_2 = float(cvd.iloc[:mid].max()), float(cvd.iloc[mid:].max())
    return {"available": True,
            "bullish": price_low_2 < price_low_1 and cvd_low_2 > cvd_low_1,
            "bearish": price_high_2 > price_high_1 and cvd_high_2 < cvd_high_1,
            "reason": None}

"""
Entry signal logic for mhr_pump_short.

Three distinct entry paths are evaluated in priority order by entry_signal():

  REENTRY        — Pump is still active; price has pulled back below the previous
                   entry; hesitation signal present.  Uses wider ATR stop.

  MHR_EXHAUSTION — N consecutive green candles form an uptrend exhaustion pattern
                   (multi-hour runner).  Normal ATR stop.

  PUMP_HESITATION — A fresh pump is detected and the current candle shows
                    momentum loss.  Uses wider ATR stop.

The returned signal dict is consumed directly by the backtest loop and by
main.py to drive live order placement.
"""

from __future__ import annotations

import pandas as pd

from .config import MHRConfig


# ═══════════════════════════════════════════════════════════════════════════
# MHR_EXHAUSTION helpers
# ═══════════════════════════════════════════════════════════════════════════

def _all_green(window: pd.DataFrame) -> bool:
    """True when every candle in the window closed above its open."""
    return bool((window["close"] > window["open"]).all())


def exhaustion_setup(window: pd.DataFrame) -> bool:
    """
    Validate an N-candle multi-hour runner exhaustion pattern.

    Conditions:
    - All candles in `window` are green (_all_green).
    - The final candle's body does not sit in the 1.3×–2.0× mean-body range.
      Rationale: a body between 1.3× and 2.0× is ambiguous — larger than a
      normal runner candle but not clearly a blow-off top.  Both extremes are
      accepted (≤ 1.3× = steady runner, ≥ 2.0× = blow-off, both are valid
      exhaustion signals).

    Returns False on an empty window or a failed green check.
    """
    if window.empty or not _all_green(window):
        return False
    bodies = (window["close"] - window["open"]).abs().astype(float)
    if len(bodies) < 2:
        return True
    mean_body = bodies.mean()
    if mean_body <= 0:
        return True
    ratio = bodies.iloc[-1] / mean_body
    if 1.3 < ratio < 2.0:
        return False
    return True


def rejection_candle(candle: pd.Series) -> tuple[bool, float]:
    """
    Determine whether a candle is a rejection candle (sellers stepping in).

    A rejection candle has an upper wick that exceeds its body, indicating
    that buyers pushed price higher but sellers pushed it back down before close.

    Returns
    -------
    (is_rejection: bool, wick_body_ratio: float)
    wick_body_ratio is inf when the body is zero (doji with any upper wick).
    """
    body = abs(float(candle["close"]) - float(candle["open"]))
    upper_wick = float(candle["high"]) - max(float(candle["open"]), float(candle["close"]))
    if body <= 0:
        ratio = float("inf") if upper_wick > 0 else 0.0
        return upper_wick > 0, ratio
    return upper_wick > body, upper_wick / body


# ═══════════════════════════════════════════════════════════════════════════
# Pump detection
# ═══════════════════════════════════════════════════════════════════════════

def detect_pump(candles: pd.DataFrame, config: MHRConfig) -> dict | None:
    """
    Identify a pump event in `candles`.

    A pump requires both a price surge and a volume surge over the lookback window:
    - Price gain ≥ config.min_pump_pct over config.lookback_hours candles.
    - Volume in the recent window ≥ config.min_volume_surge_pct × prior-window average.
      (Volume check is skipped when insufficient history is available; vol_surge = ∞.)

    Returns a metadata dict on detection, None otherwise:
      {pump_pct, volume_surge, price_24h_ago, highest_24h}
    """
    if candles.empty or len(candles) < config.lookback_hours:
        return None

    recent = candles.tail(config.lookback_hours)
    base_price = float(recent.iloc[0]["open"])
    current_price = float(recent.iloc[-1]["close"])

    if base_price <= 0:
        return None

    pump_pct = ((current_price - base_price) / base_price) * 100.0
    if pump_pct < config.min_pump_pct:
        return None

    vol_surge = float("inf")
    if "volume" in candles.columns:
        recent_vol = recent["volume"].astype(float).mean()
        if len(candles) >= config.lookback_hours * 2:
            prior = candles.iloc[-(config.lookback_hours * 2) : -config.lookback_hours]
            prior_vol = prior["volume"].astype(float).mean()
            if prior_vol > 0:
                vol_surge = recent_vol / prior_vol

    if vol_surge < config.min_volume_surge_pct:
        return None

    return {
        "pump_pct": pump_pct,
        "volume_surge": vol_surge,
        "price_24h_ago": base_price,
        "highest_24h": float(recent["high"].max()),
    }


def pump_still_active(candles: pd.DataFrame, config: MHRConfig) -> bool:
    """
    Return True when price is still ≥ 20 % above its level config.lookback_hours ago.

    Used to gate reentry attempts — we only re-enter if the pump thesis is still valid.
    """
    if candles.empty or len(candles) < config.lookback_hours:
        return False
    recent = candles.tail(config.lookback_hours)
    base = float(recent.iloc[0]["open"])
    current = float(recent.iloc[-1]["close"])
    if base <= 0:
        return False
    return ((current - base) / base) * 100.0 >= 20.0


# ═══════════════════════════════════════════════════════════════════════════
# Hesitation detection
# ═══════════════════════════════════════════════════════════════════════════

def hesitation_detected(df: pd.DataFrame, idx: int) -> bool:
    """
    Detect momentum loss on the candle at `idx` relative to the previous candle.

    Triggers on any one of the following conditions (weakest acceptable evidence):
    1. Candle is red (close < open) — directional reversal.
    2. Upper wick is present (high > max(open, close)) — sellers appeared intra-bar.
    3. Candle body is < 50 % of the prior candle's body — momentum fading.
    4. High did not exceed the prior candle's high — upward momentum stalling.

    Returns False when idx < 1 (no prior candle to compare against).
    """
    if idx < 1:
        return False

    cur = df.iloc[idx]
    prev = df.iloc[idx - 1]

    c_open = float(cur["open"])
    c_close = float(cur["close"])
    c_high = float(cur["high"])
    p_high = float(prev["high"])
    p_body = abs(float(prev["close"]) - float(prev["open"]))

    if c_close < c_open:                            # red candle
        return True
    if c_high > max(c_open, c_close):              # upper wick present
        return True
    c_body = abs(c_close - c_open)
    if p_body > 0 and c_body < p_body * 0.5:       # body shrinking
        return True
    if c_high <= p_high:                            # no new high
        return True
    return False


# ═══════════════════════════════════════════════════════════════════════════
# Unified entry filter
# ═══════════════════════════════════════════════════════════════════════════

def entry_signal(
    df: pd.DataFrame,
    idx: int,
    config: MHRConfig,
    is_reentry: bool = False,
    previous_entry_price: float = 0.0,
) -> dict | None:
    """
    Evaluate all entry paths for mhr_pump_short at candle index `idx`.

    The function requires pre-computed indicator columns in `df`:
      - ``ema20``  (optional) — used for diagnostic ema_distance_pct
      - ``rsi14``  (optional) — logged as a diagnostic field
    These are not hard gates; they are captured as metadata for post-analysis.

    Entry paths evaluated in priority order
    ----------------------------------------
    REENTRY (only when is_reentry=True):
        - pump_still_active: price ≥ 20 % above 24h-ago level
        - current open < previous_entry_price (lower entry)
        - hesitation_detected on current candle
        → uses pump_atr_sl_multiplier

    MHR_EXHAUSTION (is_reentry=False, checked first):
        - exhaustion_setup: N green candles with body-ratio validation
        → uses atr_sl_multiplier

    PUMP_HESITATION (is_reentry=False, fallback):
        - detect_pump: ≥ min_pump_pct gain + volume surge
        - hesitation_detected on current candle
        → uses pump_atr_sl_multiplier

    Parameters
    ----------
    df                    : Prepared OHLCV DataFrame with indicator columns.
    idx                   : Index of the candidate entry candle.
    config                : MHRConfig instance.
    is_reentry            : True when evaluating a reentry after a stop-out.
    previous_entry_price  : Price of the last stop-out entry (used for reentry gate).

    Returns
    -------
    Signal dict on a valid setup, None otherwise.

    Signal dict keys
    ----------------
    entry_candle_time, entry_path, is_reentry, pump_pct, volume_surge,
    ema_distance_pct, rsi14, use_pump_atr
    """
    lookback = config.lookback_green_candles
    if idx < max(lookback, config.lookback_hours):
        return None

    candle = df.iloc[idx]
    close = float(candle["close"])

    # Diagnostic fields — not hard gates
    ema_dist: float | None = None
    if "ema20" in df.columns:
        e20 = float(candle["ema20"])
        if e20 > 0:
            ema_dist = ((close - e20) / e20) * 100.0

    rsi_val: float | None = float(candle["rsi14"]) if "rsi14" in df.columns else None

    pump_window = df.iloc[max(0, idx - config.lookback_hours) : idx + 1]
    pump_info = detect_pump(pump_window, config)
    pump_pct = pump_info["pump_pct"] if pump_info else 0.0
    vol_surge = pump_info["volume_surge"] if pump_info else 0.0

    meta: dict = {
        "entry_candle_time": candle["time"],
        "pump_pct": pump_pct,
        "volume_surge": vol_surge,
        "ema_distance_pct": ema_dist,
        "rsi14": rsi_val,
    }

    # ── REENTRY ──────────────────────────────────────────────────────────
    if is_reentry:
        if not pump_still_active(pump_window, config):
            return None
        if float(candle["open"]) >= previous_entry_price:
            return None
        if not hesitation_detected(df, idx):
            return None
        return {**meta, "entry_path": "REENTRY", "is_reentry": True, "use_pump_atr": True}

    # ── MHR_EXHAUSTION ───────────────────────────────────────────────────
    setup_window = df.iloc[idx - lookback : idx]
    if exhaustion_setup(setup_window):
        return {**meta, "entry_path": "MHR_EXHAUSTION", "is_reentry": False, "use_pump_atr": False}

    # ── PUMP_HESITATION ──────────────────────────────────────────────────
    if pump_info and hesitation_detected(df, idx):
        return {**meta, "entry_path": "PUMP_HESITATION", "is_reentry": False, "use_pump_atr": True}

    return None

"""
Exit simulation for mhr_pump_short.

Implements the trailing exit model:
  - At 2R: take 50% of position (partial TP).
  - After partial fill: trail stop-loss to breakeven, then trail by (best_R - 1) × risk.
  - Max hold: exit at close of the final allowed candle.

This module is standalone — it has no dependency on config.py, entry.py, or
sizing.py.  All inputs are plain scalars and DataFrames.
"""

from __future__ import annotations

import pandas as pd


def simulate_trailing_exit(
    candles: pd.DataFrame,
    entry_idx: int,
    entry_price: float,
    initial_sl: float,
    qty: int,
    max_hold: int,
    bulk_1m: pd.DataFrame | None = None,
    use_1m_resolution: bool = True,
) -> tuple[float, float, str, object, float, int, int]:
    """
    Simulate the mhr_pump_short exit model for a SHORT position.

    Exit model
    ----------
    - Risk (R) = initial_sl − entry_price  (positive for a valid short setup)
    - Partial TP at 2R: close qty//2 (q1) when price drops 2 × risk below entry.
    - After partial fill: trail SL begins at entry (breakeven).
    - Trail rule: trail_sl = min(trail_sl, entry − (best_R − 1) × risk)
      i.e. the SL trails 1R behind the best price reached.
    - Max hold: if neither SL nor full TP is hit within max_hold candles,
      close remaining position at the last candle's close.

    When SL and TP hit on the same 1-hour candle, 1-minute data is used to
    determine which occurred first.  If 1m data is unavailable the candle
    open is used as a proximity proxy.

    Parameters
    ----------
    candles          : Prepared 1h OHLCV DataFrame (must include 'high', 'low', 'close').
    entry_idx        : Index of the entry candle in `candles`.
    entry_price      : Short entry fill price.
    initial_sl       : Initial stop-loss price (must be > entry_price for a short).
    qty              : Total contract quantity to trade.
    max_hold         : Maximum number of candles to hold the position.
    bulk_1m          : Optional 1-minute OHLCV DataFrame for intra-candle resolution.
    use_1m_resolution: Whether to use bulk_1m for SL/TP ordering; ignored if bulk_1m is None.

    Returns
    -------
    Tuple of:
      partial_pnl  : PnL from the partial TP leg (0.0 if no partial fill occurred).
      remainder_pnl: PnL from the remaining position leg.
      exit_reason  : One of the exit reason constants below.
      exit_time    : Timestamp of the closing candle.
      final_sl     : Stop-loss price at time of exit (initial_sl if never trailed).
      bars_held    : Number of candles held.
      partial_qty  : Quantity filled at partial TP (qty//2, or 0 if no partial fill).

    Exit reasons
    ------------
    SL_OR_TRAIL            — Stop hit with no prior partial fill.
    TRAIL_STOP_AFTER_PARTIAL — Stop hit after partial TP was already taken.
    MAX_HOLD_EXIT          — Max hold reached with no partial fill.
    MAX_HOLD_AFTER_PARTIAL — Max hold reached after partial fill.
    INVALID_RISK           — initial_sl ≤ entry_price; trade cannot be sized.
    """
    risk = initial_sl - entry_price
    if risk <= 0:
        return 0.0, 0.0, "INVALID_RISK", candles.iloc[entry_idx]["time"], initial_sl, 0, 0

    q1 = qty // 2       # partial TP quantity
    q2 = qty - q1       # trailing remainder quantity
    tp_price = entry_price - 2.0 * risk  # 2R target for partial fill

    partial_pnl = 0.0
    partial_done = False
    trail_sl = initial_sl
    best_r = 0.0
    bars = 0
    max_idx = min(len(candles) - 1, entry_idx + max_hold)

    for idx in range(entry_idx + 1, max_idx + 1):
        bars += 1
        c = candles.iloc[idx]
        ch, cl = float(c["high"]), float(c["low"])

        sl_hit = ch >= trail_sl
        fav_r = max(0.0, (entry_price - cl) / risk)
        tp_hit = (not partial_done) and fav_r >= 2.0

        # ── Both SL and TP hit on the same candle ────────────────────────
        if sl_hit and tp_hit:
            order = (
                _resolve_1m(c["time"], trail_sl, tp_price, bulk_1m)
                if use_1m_resolution and bulk_1m is not None
                else "UNKNOWN"
            )
            if order == "UNKNOWN":
                o = float(c["open"])
                order = "TP_FIRST" if abs(o - tp_price) < abs(o - trail_sl) else "SL_FIRST"

            if order == "TP_FIRST":
                partial_pnl = (entry_price - tp_price) * q1
                return (
                    partial_pnl,
                    (entry_price - trail_sl) * q2,
                    "TRAIL_STOP_AFTER_PARTIAL",
                    c["time"], trail_sl, bars, q1,
                )
            # SL_FIRST — no partial fill has occurred yet (tp_hit requires partial_done=False)
            return 0.0, (entry_price - trail_sl) * qty, "SL_OR_TRAIL", c["time"], trail_sl, bars, 0

        # ── SL only ──────────────────────────────────────────────────────
        if sl_hit:
            if partial_done:
                return (
                    partial_pnl,
                    (entry_price - trail_sl) * q2,
                    "TRAIL_STOP_AFTER_PARTIAL",
                    c["time"], trail_sl, bars, q1,
                )
            return 0.0, (entry_price - trail_sl) * qty, "SL_OR_TRAIL", c["time"], trail_sl, bars, 0

        # ── Update best R and trail ───────────────────────────────────────
        best_r = max(best_r, fav_r)

        if not partial_done and q1 > 0 and fav_r >= 2.0:
            partial_done = True
            partial_pnl = (entry_price - tp_price) * q1

        if best_r > 1.0:
            # Trail SL 1R behind the best price reached
            trail_sl = min(trail_sl, entry_price - (best_r - 1.0) * risk)

    # ── Max hold reached ─────────────────────────────────────────────────
    last = candles.iloc[max_idx]
    close_price = float(last["close"])
    remainder_pnl = (entry_price - close_price) * (q2 if partial_done else qty)

    if partial_done:
        return partial_pnl, remainder_pnl, "MAX_HOLD_AFTER_PARTIAL", last["time"], trail_sl, bars, q1
    return 0.0, remainder_pnl, "MAX_HOLD_EXIT", last["time"], trail_sl, bars, 0


# ── Internal helpers ──────────────────────────────────────────────────────

def _resolve_1m(
    candle_time: object,
    sl_price: float,
    tp_price: float,
    bulk_1m: pd.DataFrame,
) -> str:
    """
    Use 1-minute candle data to determine whether SL or TP was hit first
    within the hour identified by `candle_time`.

    Returns 'SL_FIRST', 'TP_FIRST', or 'UNKNOWN' (when no 1m data covers
    the period or no level was touched within it).

    Convention (short position):
      SL is hit when 1m high ≥ sl_price.
      TP is hit when 1m low  ≤ tp_price.
    """
    if bulk_1m is None or bulk_1m.empty:
        return "UNKNOWN"

    try:
        ts = candle_time.timestamp() if hasattr(candle_time, "timestamp") else float(candle_time)  # type: ignore[union-attr]
        hour_start = pd.Timestamp(ts, unit="s", tz="UTC")
    except Exception:
        return "UNKNOWN"

    hour_end = hour_start + pd.Timedelta(hours=1)
    mask = (bulk_1m["time"] >= hour_start) & (bulk_1m["time"] < hour_end)
    within = bulk_1m[mask]

    if within.empty:
        return "UNKNOWN"

    for _, mc in within.iterrows():
        h, l = float(mc["high"]), float(mc["low"])
        sl_hit = h >= sl_price
        tp_hit = l <= tp_price
        if sl_hit and tp_hit:
            o = float(mc["open"])
            return "TP_FIRST" if abs(o - tp_price) < abs(o - sl_price) else "SL_FIRST"
        if sl_hit:
            return "SL_FIRST"
        if tp_hit:
            return "TP_FIRST"

    return "UNKNOWN"

"""
Technical indicator functions used by mhr_pump_short.

All functions are pure — they take pandas Series / DataFrames and return
Series or scalar values without side effects.
"""

import pandas as pd


def ema(series: pd.Series, length: int) -> pd.Series:
    """Exponential moving average (EMA) using Wilder / ewm span convention."""
    return series.ewm(span=length, adjust=False).mean()


def rsi(series: pd.Series, period: int = 14) -> pd.Series:
    """
    Relative Strength Index (RSI) using Wilder's smoothed averages.

    Warm-up candles (first `period` rows) are set to the neutral value 50.0
    so that downstream filters are not triggered by uninitialised data.
    """
    delta = series.diff()
    gain = delta.clip(lower=0).ewm(alpha=1 / period, adjust=False).mean()
    loss = (-delta.clip(upper=0)).ewm(alpha=1 / period, adjust=False).mean()

    result = pd.Series(50.0, index=series.index, dtype=float)
    both = (gain > 0) & (loss > 0)
    result[both] = 100.0 - (100.0 / (1.0 + gain[both] / loss[both]))
    result[(gain > 0) & (loss == 0)] = 100.0
    result[(gain == 0) & (loss > 0)] = 0.0
    result.iloc[:period] = 50.0
    return result


def atr(df: pd.DataFrame, period: int = 14) -> pd.Series:
    """
    Average True Range (ATR) using a simple rolling mean of True Range.

    True Range = max(high-low, |high-prev_close|, |low-prev_close|).
    """
    high = df["high"]
    low = df["low"]
    prev_close = df["close"].shift(1)
    true_range = pd.concat(
        [high - low, (high - prev_close).abs(), (low - prev_close).abs()],
        axis=1,
    ).max(axis=1)
    return true_range.rolling(window=period, min_periods=1).mean()


def candle_volume_usd(candles: pd.DataFrame, lookback_bars: int = 24) -> float:
    """
    Approximate USD volume over the last `lookback_bars` candles.

    Uses `volume × close` when a close column is present; otherwise sums
    raw volume.  Returns 0.0 if volume data is absent or the DataFrame is empty.
    """
    if candles.empty or "volume" not in candles.columns:
        return 0.0
    recent = candles.tail(lookback_bars)
    if "close" in candles.columns:
        return float((recent["volume"] * recent["close"]).sum())
    return float(recent["volume"].sum())

"""
scalp_pullback strategy — 5-minute trend-following micro pullback.

Execution timeframe : 5-minute candles (resampled from 1m if needed)
Trend filter        : 1-hour EMA50 (resampled from 5m)

def resample_to_5m(df_1m: pd.DataFrame) -> pd.DataFrame:
    """Resample 1-minute OHLCV to 5-minute candles."""
    if df_1m.empty:
        return pd.DataFrame()
    df = df_1m.copy().set_index("time")
    r = df.resample("5min").agg({
        "open": "first", "high": "max",
        "low": "min", "close": "last", "volume": "sum",
    }).dropna().reset_index()
    return r


def resample_5m_to_1h(df_5m: pd.DataFrame) -> pd.DataFrame:
    """Resample 5-minute candles to 1-hour for trend filter."""
    if df_5m.empty:
        return pd.DataFrame()
    df = df_5m.copy().set_index("time")
    r = df.resample("1h").agg({
        "open": "first", "high": "max",
        "low": "min", "close": "last", "volume": "sum",
    }).dropna().reset_index()
    return r

def _has_above_avg_volume(prepared: pd.DataFrame, idx: int, lookback: int) -> bool:
    """Signal candle must have above-average volume."""
    if "volume" not in prepared.columns or idx < lookback:
        return True
    current = float(prepared.iloc[idx]["volume"])
    recent = prepared["volume"].iloc[max(0, idx - lookback):idx].astype(float)
    if recent.empty or recent.mean() <= 0:
        return True
    return current >= recent.mean()


def detect_long(prepared: pd.DataFrame, idx: int, config: ScalpConfig) -> dict | None:
    """
    LONG setup on 5-minute candles.

    Requirements:
    - Signal candle is green
    - Price above EMA20(5m)
    - 3+ consecutive red candles before (pullback)
    - Signal candle has above-average volume
    - SL below pullback low with buffer
    """
    if idx < config.max_pullback_candles + 1:
        return None

    sig = prepared.iloc[idx]

    # Signal candle must be green
    if float(sig["close"]) <= float(sig["open"]):
        return None

    # Price must be above EMA20
    if "ema20" in prepared.columns and float(sig["close"]) <= float(sig["ema20"]):
        return None

    atr_val = float(sig["atr14"]) if "atr14" in prepared.columns else 0

    # Volume confirmation
    if not _has_above_avg_volume(prepared, idx, config.vol_lookback):
        return None

    # Count consecutive red candles before signal
    count = 0
    low = float("inf")
    for lb in range(1, config.max_pullback_candles + 1):
        if idx - lb < 0:
            break
        c = prepared.iloc[idx - lb]
        if float(c["close"]) < float(c["open"]):
            count += 1
            low = min(low, float(c["low"]))
        else:
            break

    if count < config.min_pullback_candles:
        return None

    # SL below pullback low with buffer
    sl = low * (1.0 - config.sl_buffer_pct)

    return {
        "direction": "LONG",
        "sl_price": sl,
        "pullback_candles": count,
        "atr": atr_val,
        "signal_candle_time": sig["time"],
    }


def detect_short(prepared: pd.DataFrame, idx: int, config: ScalpConfig) -> dict | None:
    """
    SHORT setup on 5-minute candles.

    Requirements:
    - Signal candle is red
    - Price below EMA20(5m)
    - 3+ consecutive green candles before (pullback)
    - Signal candle has above-average volume
    - SL above pullback high with buffer
    """
    if idx < config.max_pullback_candles + 1:
        return None

    sig = prepared.iloc[idx]

    if float(sig["close"]) >= float(sig["open"]):
        return None

    if "ema20" in prepared.columns and float(sig["close"]) >= float(sig["ema20"]):
        return None

    atr_val = float(sig["atr14"]) if "atr14" in prepared.columns else 0

    if not _has_above_avg_volume(prepared, idx, config.vol_lookback):
        return None

    count = 0
    high = 0.0
    for lb in range(1, config.max_pullback_candles + 1):
        if idx - lb < 0:
            break
        c = prepared.iloc[idx - lb]
        if float(c["close"]) > float(c["open"]):
            count += 1
            high = max(high, float(c["high"]))
        else:
            break

    if count < config.min_pullback_candles:
        return None

    sl = high * (1.0 + config.sl_buffer_pct)

    return {
        "direction": "SHORT",
        "sl_price": sl,
        "pullback_candles": count,
        "atr": atr_val,
        "signal_candle_time": sig["time"],
    }


def scalp_entry(prepared: pd.DataFrame, idx: int, trend: str,
                config: ScalpConfig) -> dict | None:
    """Dispatch to long or short detection based on 1h trend."""
    if idx < max(config.ema_5m_length, config.max_pullback_candles + 2):
        return None

    if config.min_volume > 0 and "volume" in prepared.columns:
        if float(prepared.iloc[idx]["volume"]) < config.min_volume:
            return None

    if trend == "BULLISH":
        return detect_long(prepared, idx, config)
    if trend == "BEARISH":
        return detect_short(prepared, idx, config)
    return None

def scalp_qty(entry_price: float, risk: float, config: ScalpConfig,
              override_capital: float | None = None) -> float:
    """
    Compute position quantity for a scalp trade.

    Returns fractional qty (rounded to 3 decimals) for high-price tokens like BTC.
    """
    capital = override_capital or config.capital_deployed
    if capital <= 0 or entry_price <= 0 or risk <= 0:
        return 0
    max_notional = capital * config.leverage
    sl_loss = capital * config.sl_capital_pct
    qty = min(sl_loss / risk, max_notional / entry_price)
    qty = round(qty, 3)
    return min(qty, 10_000_000) if qty > 0 else 0


def scalp_trade_capital(available: float, config: ScalpConfig) -> float:
    """Allocate capital for the next scalp trade (same logic as MHR)."""
    tc = available * config.max_capital_usage_pct
    return tc if tc >= 1.0 else 0.0


def cap_scalp_pnl(net_pnl: float, trade_cap: float, config: ScalpConfig) -> float:
    """Clamp PnL to prevent unrealistic equity swings."""
    return max(min(net_pnl, trade_cap * config.max_pnl_impact_pct), -trade_cap)


# ═══════════════════════════════════════════════════════════════════════════════
# EXIT SIMULATION
# ═══════════════════════════════════════════════════════════════════════════════

def simulate_scalp_exit(
    candles: pd.DataFrame,
    entry_idx: int,
    entry_price: float,
    sl_price: float,
    qty: float,
    direction: str,
    config: ScalpConfig,
) -> tuple[float, str, object, float, int]:
    """
    Simulate scalp exit: 50% at TP1, trail remaining at best_r - 1R.

    Returns (pnl, exit_reason, exit_time, exit_price, bars_held).
    """
    risk = abs(entry_price - sl_price) if direction == "LONG" else abs(sl_price - entry_price)
    if risk <= 0:
        return 0.0, "INVALID_RISK", candles.iloc[entry_idx]["time"], entry_price, 0

    q1 = round(qty * config.tp1_qty_pct, 3)
    q2 = round(qty - q1, 3)
    partial = 0.0
    partial_done = False
    trail_sl = sl_price
    best_r = 0.0
    bars = 0
    max_idx = min(len(candles) - 1, entry_idx + config.max_holding_candles)

    for idx in range(entry_idx + 1, max_idx + 1):
        bars += 1
        c = candles.iloc[idx]
        ch, cl = float(c["high"]), float(c["low"])

        if direction == "LONG":
            if cl <= trail_sl:
                pnl = partial + (trail_sl - entry_price) * (q2 if partial_done else qty)
                return pnl, "TRAIL_SL" if partial_done else "SL_HIT", c["time"], trail_sl, bars
            fav_r = (ch - entry_price) / risk if risk > 0 else 0
        else:
            if ch >= trail_sl:
                pnl = partial + (entry_price - trail_sl) * (q2 if partial_done else qty)
                return pnl, "TRAIL_SL" if partial_done else "SL_HIT", c["time"], trail_sl, bars
            fav_r = (entry_price - cl) / risk if risk > 0 else 0

        best_r = max(best_r, fav_r)

        if not partial_done and q1 > 0 and fav_r >= config.tp1_rr:
            partial_done = True
            partial = risk * config.tp1_rr * q1
            trail_sl = entry_price

        if partial_done and best_r > config.trail_lock_rr:
            lock = best_r - config.trail_lock_rr
            if direction == "LONG":
                trail_sl = max(trail_sl, entry_price + lock * risk)
            else:
                trail_sl = min(trail_sl, entry_price - lock * risk)

    last = candles.iloc[max_idx]
    ep = float(last["close"])
    if direction == "LONG":
        pnl = partial + (ep - entry_price) * (q2 if partial_done else qty)
    else:
        pnl = partial + (entry_price - ep) * (q2 if partial_done else qty)
    return pnl, "MAX_HOLD_PARTIAL" if partial_done else "MAX_HOLD", last["time"], ep, bars


# ═══════════════════════════════════════════════════════════════════════════════
# BACKTEST RUNNER
# ═══════════════════════════════════════════════════════════════════════════════

def backtest_scalp(
    symbol: str,
    candles_raw: pd.DataFrame,
    config: ScalpConfig,
    running_equity: float | None = None,
    charge_rate: float = 0.0005,
) -> tuple[list[dict], float]:
    """
    Run scalp backtest on 5-minute candles (auto-resampled from 1m if needed).

    Returns (trades_list, final_equity).
    """
    trades: list[dict] = []
    equity = running_equity or config.capital_deployed

    if candles_raw.empty or len(candles_raw) < 300:
        logger.info(f"  {symbol} | scalp: too few candles ({len(candles_raw)})")
        return trades, equity

    # Auto-detect 1m vs 5m input and resample if needed
    sorted_raw = candles_raw.copy().sort_values("time").reset_index(drop=True)
    time_diff = 60
    if len(sorted_raw) >= 2:
        time_diff = (sorted_raw["time"].iloc[1] - sorted_raw["time"].iloc[0]).total_seconds()

    if time_diff < 120:
        df_5m = resample_to_5m(sorted_raw)
    else:
        df_5m = sorted_raw

    if df_5m.empty or len(df_5m) < 300:
        logger.info(f"  {symbol} | scalp: not enough 5m candles ({len(df_5m)})")
        return trades, equity

    # Prepare 5m indicators
    prepared = df_5m.copy().sort_values("time").reset_index(drop=True)
    prepared["ema20"] = ema(prepared["close"].astype(float), config.ema_5m_length)
    prepared["atr14"] = atr(prepared, 14)

    # Build 1h candles for trend filter
    df_1h = resample_5m_to_1h(prepared)
    if df_1h.empty or len(df_1h) < config.ema_1h_length + 1:
        logger.info(f"  {symbol} | scalp: not enough 1h ({len(df_1h)})")
        return trades, equity

    df_1h["ema50"] = ema(df_1h["close"].astype(float), config.ema_1h_length)

    # Build sorted trend lookup for O(log n) bisect search
    sorted_1h_times = df_1h["time"].tolist()
    sorted_1h_trends = []
    for _, row in df_1h.iterrows():
        p = float(row["close"])
        e = float(row["ema50"])
        sorted_1h_trends.append("BULLISH" if e > 0 and p > e else ("BEARISH" if e > 0 else None))

    def get_trend(ts):
        pos = bisect_right(sorted_1h_times, ts) - 1
        return sorted_1h_trends[pos] if 0 <= pos < len(sorted_1h_trends) else None

    # ── Scan loop ────────────────────────────────────────────────────────
    idx = max(config.max_pullback_candles + 2, config.ema_5m_length + 5)
    session_trades = 0
    last_trade_idx = -config.cooldown_candles - 1
    stats = {"trend_none": 0, "trend_bull": 0, "trend_bear": 0,
             "filter_rej": 0, "cap_rej": 0, "qty_rej": 0, "risk_rej": 0,
             "longs": 0, "shorts": 0}

    while idx < len(prepared) - 1:
        if idx - last_trade_idx < config.cooldown_candles:
            idx += 1; continue
        if session_trades >= config.max_trades_per_session:
            break

        trend = get_trend(prepared.iloc[idx]["time"])
        if not trend:
            stats["trend_none"] += 1; idx += 1; continue

        stats["trend_bull" if trend == "BULLISH" else "trend_bear"] += 1

        setup = scalp_entry(prepared, idx, trend, config)
        if not setup:
            stats["filter_rej"] += 1; idx += 1; continue

        direction = setup["direction"]
        sl_price = setup["sl_price"]

        # Direct entry at next candle open
        nxt = idx + 1
        if nxt >= len(prepared):
            break
        entry_price = float(prepared.iloc[nxt]["open"])
        risk = abs(entry_price - sl_price) if direction == "LONG" else abs(sl_price - entry_price)

        if risk <= 0:
            stats["risk_rej"] += 1; idx += 1; continue

        trade_cap = scalp_trade_capital(equity, config)
        if trade_cap < 1.0:
            stats["cap_rej"] += 1; idx += 1; continue

        qty = scalp_qty(entry_price, risk, config, trade_cap)
        if qty <= 0:
            stats["qty_rej"] += 1; idx += 1; continue

        pnl, reason, exit_time, exit_price, bars = simulate_scalp_exit(
            prepared, nxt, entry_price, sl_price, qty, direction, config,
        )

        fees = calculate_fees(symbol, entry_price, exit_price, qty,
                              charge_rate, prepared.iloc[idx]["time"], exit_time)
        net = pnl - fees
        net = cap_scalp_pnl(net, trade_cap, config)

        eq_before = equity
        equity += net
        if equity < 0:
            equity = 0.0

        stats["longs" if direction == "LONG" else "shorts"] += 1
        session_trades += 1

        trades.append({
            "symbol": symbol, "strategy": "scalp_pullback",
            "direction": direction,
            "entry_time": prepared.iloc[idx]["time"], "exit_time": exit_time,
            "entry_price": entry_price, "sl_price": sl_price,
            "exit_price": exit_price,
            "qty": qty, "trade_capital": trade_cap,
            "equity_before_trade": eq_before, "equity_after_trade": equity,
            "pullback_candles": setup["pullback_candles"],
            "atr14": setup["atr"], "trend_5m": trend,
            "bars_held": bars, "gross_pnl": pnl, "fees": fees,
            "net_pnl": net, "rr_achieved": pnl / (risk * qty) if risk * qty > 0 else 0,
            "roi_on_capital": net / trade_cap if trade_cap > 0 else 0,
            "exit_reason": reason,
            "entry_path": f"SCALP_{direction}",
            "is_reentry": False, "pump_pct": 0, "volume_surge": 0,
            "ema_distance_pct": None, "rsi14": None,
            "wick_body_ratio": None, "target_price": 0,
            "initial_sl_price": sl_price, "final_sl_price": sl_price,
            "atr_pct": 0, "partial_qty_filled": 0,
            "lower_high_price": None, "exhaustion_5m_signals": None,
            "rsi_5m": None, "sl_method": "pullback",
        })

        last_trade_idx = nxt + bars
        idx = last_trade_idx + config.cooldown_candles
        if equity <= 0:
            break

    logger.info(
        f"  {symbol} | scalp_5m: "
        f"5m={len(prepared)}, 1h={len(df_1h)}, "
        f"bull={stats['trend_bull']}, bear={stats['trend_bear']}, "
        f"rej={stats['filter_rej']}, "
        f"L={stats['longs']}, S={stats['shorts']}, "
        f"trades={len(trades)}, eq={equity:.2f}"
    )
    return trades, equity

"""
Position sizing for mhr_pump_short.

Three public functions cover the full sizing lifecycle:
  trade_capital  — how much equity to deploy on the next trade
  sl_tp_qty      — stop-loss price, take-profit price, and contract quantity
  cap_pnl        — post-simulation PnL clamp to prevent unrealistic equity swings
"""

from __future__ import annotations

import math

from .config import MHRConfig


def trade_capital(available: float, config: MHRConfig) -> float:
    """
    Return the capital to deploy on the next trade.

    Allocates `max_capital_usage_pct` of available equity.  Returns 0.0 when
    the resulting amount is below 1.0 (too small to size a position).
    """
    allocated = available * config.max_capital_usage_pct
    return allocated if allocated >= 1.0 else 0.0


def sl_tp_qty(
    entry_price: float,
    config: MHRConfig,
    capital: float | None = None,
    atr_value: float | None = None,
    use_pump_atr: bool = False,
) -> tuple[float, float, int]:
    """
    Compute stop-loss price, take-profit price, and contract quantity for a short entry.

    Sizing logic
    ------------
    1. Stop-loss distance is ATR-based when atr_value is provided:
           sl_dist = atr_value × (pump_atr_sl_multiplier if use_pump_atr else atr_sl_multiplier)
       Quantity is the smaller of:
           - max loss budget / sl_dist   (keeps max loss ≤ sl_capital_pct × capital)
           - max notional / entry_price  (keeps position within leverage limit)
       Falls back to filling the full notional when no ATR is available.

    2. Take-profit distance satisfies both:
           tp_target = max(pt_capital_pct × capital, min_rr × sl_loss)
           tp_price  = entry_price − tp_target / qty

    3. Quantity is floored to the nearest integer and hard-capped at 10,000,000.

    Parameters
    ----------
    entry_price  : Short entry price (must be > 0).
    config       : MHRConfig instance.
    capital      : Override for trade capital; falls back to config.capital_deployed.
    atr_value    : Current ATR value; None triggers fallback to full-notional sizing.
    use_pump_atr : When True, use the wider pump ATR multiplier.

    Returns
    -------
    (sl_price, tp_price, qty)

    Raises
    ------
    ValueError if entry_price ≤ 0, capital ≤ 0, or the resulting quantity < 1.
    """
    if entry_price <= 0:
        raise ValueError(f"entry_price must be positive, got {entry_price}")
    cap = capital if capital is not None else config.capital_deployed
    if cap <= 0:
        raise ValueError(f"capital must be positive, got {cap}")

    max_notional = cap * config.leverage
    sl_loss = cap * config.sl_capital_pct
    multiplier = config.pump_atr_sl_multiplier if use_pump_atr else config.atr_sl_multiplier

    if atr_value and atr_value > 0:
        sl_dist = atr_value * multiplier
        raw_qty = min(sl_loss / sl_dist, max_notional / entry_price)
    else:
        raw_qty = max_notional / entry_price
        sl_dist = None

    qty = min(math.floor(raw_qty), 10_000_000)
    if qty < 1:
        raise ValueError("computed quantity < 1 — trade capital or ATR too small")

    # Guard against floating-point overshoot of notional limit
    if qty * entry_price > max_notional * 1.01:
        qty = math.floor(max_notional / entry_price)
        if qty < 1:
            raise ValueError("quantity < 1 after notional cap")

    sl_price = entry_price + (sl_dist if sl_dist is not None else sl_loss / qty)
    tp_target = max(cap * config.pt_capital_pct, sl_loss * config.min_rr)
    tp_price = entry_price - (tp_target / qty)

    return float(sl_price), float(tp_price), int(qty)


def structural_sl_tp_qty(
    entry_price: float,
    prev_high: float,
    config: MHRConfig,
    capital: float | None = None,
    atr_value: float | None = None,
    use_pump_atr: bool = False,
) -> tuple[float, float, int]:
    """
    Compute SL, TP, and qty using the previous candle high as the stop level.

    SL is placed at prev_high (no buffer).
    qty = floor(max_loss / (prev_high − entry_price)).

    Falls back to sl_tp_qty() (ATR-based) when:
      - prev_high ≤ entry_price (gap-up; structure is below entry)
      - resulting qty < 1 (candle wick too wide for the given capital)

    Parameters
    ----------
    entry_price  : Short entry price.
    prev_high    : High of the candle immediately before the entry candle.
    config       : MHRConfig instance.
    capital      : Trade capital override; defaults to config.capital_deployed.
    atr_value    : Current ATR value passed through to fallback sl_tp_qty().
    use_pump_atr : Passed through to fallback sl_tp_qty().

    Returns
    -------
    (sl_price, tp_price, qty)

    Raises
    ------
    ValueError if both structural and ATR sizing fail to produce qty ≥ 1.
    """
    if entry_price <= 0:
        raise ValueError(f"entry_price must be positive, got {entry_price}")
    cap = capital if capital is not None else config.capital_deployed
    if cap <= 0:
        raise ValueError(f"capital must be positive, got {cap}")

    if prev_high > entry_price:
        sl_dist = prev_high - entry_price
        max_loss = cap * config.sl_capital_pct
        raw_qty = math.floor(max_loss / sl_dist)
        if raw_qty >= 1:
            qty = min(raw_qty, math.floor(cap * config.leverage / entry_price))
            if qty >= 1:
                tp_target = max(cap * config.pt_capital_pct, max_loss * config.min_rr)
                tp_price = entry_price - tp_target / qty
                return float(prev_high), float(tp_price), int(qty)

    return sl_tp_qty(entry_price, config, capital, atr_value, use_pump_atr)


def cap_pnl(net_pnl: float, trade_cap: float, config: MHRConfig) -> float:
    """
    Clamp net PnL to [-trade_cap, trade_cap × max_pnl_impact_pct].

    Prevents a single simulated trade from swinging equity beyond what the
    leverage and capital limits would allow in live conditions.
    """
    return max(min(net_pnl, trade_cap * config.max_pnl_impact_pct), -trade_cap)

"""
Dynamic fee model for mhr_pump_short.

Delta Exchange charges a single-side maker/taker fee.  For very short-duration
trades the exit leg may be waived or negligible, so the model distinguishes
between a fast-exit (entry fee only) and a normal round-trip.

Fast-exit thresholds
--------------------
BTC / ETH pairs : 30 minutes  (higher liquidity → quicker fills)
All other coins : 15 minutes
"""

from __future__ import annotations


BTC_ETH_SYMBOLS: frozenset[str] = frozenset({
    "BTCUSD", "ETHUSD", "BTCUSDT", "ETHUSDT",
    "BTCPERP", "ETHPERP", "XBTUSD",
})

_FAST_THRESHOLD_BTC_ETH: float = 30.0   # minutes
_FAST_THRESHOLD_OTHER: float = 15.0     # minutes
_SINGLE_SIDE_RATE: float = 0.0005       # 0.05 %


def calculate_fees(
    symbol: str,
    entry_price: float,
    exit_price: float,
    qty: float,
    charge_rate: float,
    entry_time: object,
    exit_time: object,
) -> float:
    """
    Return the total fee for a completed trade.

    Parameters
    ----------
    symbol       : Trading symbol (used to select the fast-exit threshold).
    entry_price  : Fill price at trade open.
    exit_price   : Fill price at trade close.
    qty          : Contract quantity.
    charge_rate  : Round-trip fee rate (e.g. 0.0005 for 0.05 %).
    entry_time   : Trade open timestamp — datetime, pd.Timestamp, or Unix float.
    exit_time    : Trade close timestamp — same type as entry_time.

    Returns
    -------
    Fee in the same currency as price × qty (i.e. USD for USD-margined contracts).
    """
    fast_threshold = (
        _FAST_THRESHOLD_BTC_ETH
        if symbol.upper() in BTC_ETH_SYMBOLS
        else _FAST_THRESHOLD_OTHER
    )
    duration = _duration_minutes(entry_time, exit_time)
    entry_notional = abs(entry_price * qty)
    exit_notional = abs(exit_price * qty)

    if duration <= fast_threshold:
        return entry_notional * _SINGLE_SIDE_RATE
    return (entry_notional + exit_notional) * charge_rate


# ── Internal helpers ──────────────────────────────────────────────────────

def _duration_minutes(entry_time: object, exit_time: object) -> float:
    """Convert two timestamps to a duration in minutes.  Returns inf on failure."""
    try:
        if hasattr(entry_time, "timestamp") and hasattr(exit_time, "timestamp"):
            return (exit_time.timestamp() - entry_time.timestamp()) / 60.0  # type: ignore[union-attr]
        if isinstance(entry_time, (int, float)) and isinstance(exit_time, (int, float)):
            return (float(exit_time) - float(entry_time)) / 60.0
    except Exception:
        pass
    return float("inf")

import math
import numpy as np
from dataclasses import dataclass
from typing import Optional

import pandas as pd


# ═══════════════════════════════════════════════════════════════════════
# MOVING AVERAGE IMPLEMENTATIONS
# All 12 MA types from the PineScript f_ma() function
# ═══════════════════════════════════════════════════════════════════════

def ma_sma(series: pd.Series, length: int) -> pd.Series:
    """Simple Moving Average."""
    return series.rolling(window=length, min_periods=1).mean()


def ma_ema(series: pd.Series, length: int) -> pd.Series:
    """Exponential Moving Average."""
    return series.ewm(span=length, adjust=False).mean()


def ma_rma(series: pd.Series, length: int) -> pd.Series:
    """
    Running Moving Average (Wilder's smoothing).
    Equivalent to ta.rma() in PineScript.
    alpha = 1/length
    """
    return series.ewm(alpha=1.0 / length, adjust=False).mean()


def ma_wma(series: pd.Series, length: int) -> pd.Series:
    """
    Weighted Moving Average.
    Equivalent to ta.wma() in PineScript.
    """
    weights = np.arange(1, length + 1, dtype=float)

    def _wma(window):
        w = weights[-len(window):]
        return np.dot(window, w) / w.sum()

    return series.rolling(window=length, min_periods=length).apply(_wma, raw=True)


def ma_vwma(df: pd.DataFrame, length: int) -> pd.Series:
    """
    Volume Weighted Moving Average.
    Equivalent to ta.vwma() in PineScript.
    Falls back to SMA if no volume column.
    """
    if "volume" not in df.columns:
        return ma_sma(df["close"].astype(float), length)
    close = df["close"].astype(float)
    vol = df["volume"].astype(float)
    return (close * vol).rolling(window=length, min_periods=1).sum() / \
           vol.rolling(window=length, min_periods=1).sum().replace(0, float("nan"))


def ma_tema(series: pd.Series, length: int) -> pd.Series:
    """
    Triple Exponential Moving Average.
    PineScript: 3 * ema1 - 3 * ema2 + ema3
    """
    ema1 = ma_ema(series, length)
    ema2 = ma_ema(ema1, length)
    ema3 = ma_ema(ema2, length)
    return 3.0 * ema1 - 3.0 * ema2 + ema3


def ma_dema(series: pd.Series, length: int) -> pd.Series:
    """
    Double Exponential Moving Average.
    PineScript: 2 * ema(src, len) - ema(ema(src, len), len)
    """
    ema1 = ma_ema(series, length)
    ema2 = ma_ema(ema1, length)
    return 2.0 * ema1 - ema2


def ma_tma(series: pd.Series, length: int) -> pd.Series:
    """
    Triangular Moving Average.
    PineScript: sma(sma(src, len), len)
    """
    sma1 = ma_sma(series, length)
    return ma_sma(sma1, length)


def ma_alma(series: pd.Series, length: int, offset: float = 0.85,
            sigma: float = 6.0) -> pd.Series:
    """
    Arnaud Legoux Moving Average.
    Equivalent to ta.alma(src, len, 0.85, 6) in PineScript.
    """
    m = math.floor(offset * (length - 1))
    s = length / sigma
    weights = np.array([
        math.exp(-((i - m) ** 2) / (2 * s * s))
        for i in range(length)
    ])
    weights = weights / weights.sum()

    def _alma(window):
        if len(window) < length:
            return float("nan")
        return np.dot(window, weights)

    return series.rolling(window=length, min_periods=length).apply(_alma, raw=True)


def ma_hma(series: pd.Series, length: int) -> pd.Series:
    """
    Hull Moving Average.
    Equivalent to ta.hma() in PineScript.
    HMA = WMA(2 * WMA(src, len/2) - WMA(src, len), sqrt(len))
    """
    half_len = max(1, length // 2)
    sqrt_len = max(1, int(math.sqrt(length)))
    wma_half = ma_wma(series, half_len)
    wma_full = ma_wma(series, length)
    diff = 2.0 * wma_half - wma_full
    # Fill NaN before computing final WMA
    diff = diff.fillna(method="bfill")
    return ma_wma(diff, sqrt_len)


def ma_lsma(series: pd.Series, length: int) -> pd.Series:
    """
    Least Squares Moving Average (Linear Regression).
    Equivalent to ta.linreg(src, len, 0) in PineScript.
    """
    def _linreg(window):
        if len(window) < length:
            return float("nan")
        x = np.arange(length, dtype=float)
        y = np.array(window, dtype=float)
        # Linear regression: y = mx + b, return value at x = length-1
        n = len(x)
        sum_x = x.sum()
        sum_y = y.sum()
        sum_xy = (x * y).sum()
        sum_x2 = (x * x).sum()
        denom = n * sum_x2 - sum_x * sum_x
        if denom == 0:
            return y[-1]
        m = (n * sum_xy - sum_x * sum_y) / denom
        b = (sum_y - m * sum_x) / n
        return m * (n - 1) + b  # value at offset 0

    return series.rolling(window=length, min_periods=length).apply(_linreg, raw=True)


def ma_zlema(series: pd.Series, length: int) -> pd.Series:
    """
    Zero Lag Exponential Moving Average.
    PineScript:
        lag = floor((len - 1) / 2)
        priceAdjusted = src + (src - src[lag])
        result = ema(priceAdjusted, len)
    """
    lag = (length - 1) // 2
    adjusted = series + (series - series.shift(lag))
    adjusted = adjusted.fillna(series)
    return ma_ema(adjusted, length)


def get_ma(df: pd.DataFrame, ma_type: str, length: int) -> pd.Series:
    """
    Master MA selector — matches the PineScript f_ma() function exactly.
    Supports all 12 MA types from the original script.
    """
    series = df["close"].astype(float)
    ma_type = ma_type.upper()

    ma_map = {
        "EMA": lambda: ma_ema(series, length),
        "SMA": lambda: ma_sma(series, length),
        "RMA": lambda: ma_rma(series, length),
        "WMA": lambda: ma_wma(series, length),
        "VWMA": lambda: ma_vwma(df, length),
        "TEMA": lambda: ma_tema(series, length),
        "DEMA": lambda: ma_dema(series, length),
        "TMA": lambda: ma_tma(series, length),
        "ALMA": lambda: ma_alma(series, length),
        "HMA": lambda: ma_hma(series, length),
        "LSMA": lambda: ma_lsma(series, length),
        "ZLEMA": lambda: ma_zlema(series, length),
    }

    if ma_type not in ma_map:
        raise ValueError(
            f"Unknown MA type: {ma_type}. "
            f"Supported: {', '.join(sorted(ma_map.keys()))}"
        )

    return ma_map[ma_type]()


# ═══════════════════════════════════════════════════════════════════════
# ATR FOR STOP LOSS SIZING
# ═══════════════════════════════════════════════════════════════════════

def calculate_atr(df: pd.DataFrame, period: int = 14) -> pd.Series:
    high = df["high"].astype(float)
    low = df["low"].astype(float)
    prev_close = df["close"].astype(float).shift(1)
    tr = pd.concat(
        [(high - low), (high - prev_close).abs(), (low - prev_close).abs()],
        axis=1,
    ).max(axis=1)
    return tr.rolling(window=period, min_periods=1).mean()

@dataclass(frozen=True)
class FastSlowMAConfig:

    fast_length: int = 24
    slow_length: int = 65
    ma_type: str = "EMA"

    # Trade direction
    allow_long: bool = True
    allow_short: bool = False       # useShorts = false in PineScript

    # Capital / risk management
    capital_deployed: float = 100.0
    leverage: float = 20.0

    min_entry_price: float = 0.10
    max_pnl_impact_pct: float = 2.0

    # ATR-based SL/TP
    atr_sl_multiplier: float = 1.5

FAST_SLOW_DEFAULT_CONFIG = FastSlowMAConfig()


# ═══════════════════════════════════════════════════════════════════════
# CROSSOVER / CROSSUNDER DETECTION
#
# Matches PineScript:
#   ta.crossover(maMomentum, 0)  → fast crosses above slow
#   ta.crossunder(maMomentum, 0) → fast crosses below slow
# ═══════════════════════════════════════════════════════════════════════

def detect_crossover(fast: pd.Series, slow: pd.Series) -> pd.Series:
    """Bullish crossover: fast crosses ABOVE slow."""
    return ((fast > slow) & (fast.shift(1) <= slow.shift(1))).fillna(False)


def detect_crossunder(fast: pd.Series, slow: pd.Series) -> pd.Series:
    """Bearish crossunder: fast crosses BELOW slow."""
    return ((fast < slow) & (fast.shift(1) >= slow.shift(1))).fillna(False)

def calculate_trade_capital(available_capital: float, config: FastSlowMAConfig) -> float:
    """Strictly 50% of available capital."""
    tc = available_capital * config.max_capital_usage_pct
    return tc if tc >= 1.0 else 0.0


def calculate_long_sl_tp(
    entry_price: float,
    config: FastSlowMAConfig,
    override_capital: float | None = None,
    atr_value: float | None = None,
) -> tuple[float, float, int]:
    """
    Compute SL, TP and quantity for a LONG position.
    SL is BELOW entry, TP is ABOVE entry.
    """
    if entry_price <= 0:
        raise ValueError("entry_price must be > 0")
    if entry_price < config.min_entry_price:
        raise ValueError(f"price {entry_price:.6f} below {config.min_entry_price}")

    capital = override_capital if override_capital is not None else config.capital_deployed
    if capital <= 0:
        raise ValueError("capital must be > 0")

    max_notional = capital * config.leverage
    sl_loss = capital * config.sl_capital_pct

    if atr_value and atr_value > 0:
        sl_dist = atr_value * config.atr_sl_multiplier
        raw_qty = min(sl_loss / sl_dist, max_notional / entry_price)
    else:
        raw_qty = max_notional / entry_price
        sl_dist = None

    qty = min(math.floor(raw_qty), config.max_quantity)
    if qty < 1:
        raise ValueError("quantity < 1")
    if qty * entry_price > max_notional * 1.01:
        qty = math.floor(max_notional / entry_price)
        if qty < 1:
            raise ValueError("quantity < 1 after cap")

    # LONG: SL below entry, TP above entry
    if sl_dist is not None:
        sl_price = entry_price - sl_dist
        tp_price = entry_price + sl_dist * config.tp_rr_ratio
    else:
        sl_price = entry_price - (sl_loss / qty)
        tp_price = entry_price + (sl_loss / qty) * config.tp_rr_ratio

    return float(max(sl_price, 0.0)), float(tp_price), int(qty)


def calculate_short_sl_tp(
    entry_price: float,
    config: FastSlowMAConfig,
    override_capital: float | None = None,
    atr_value: float | None = None,
) -> tuple[float, float, int]:
    """
    Compute SL, TP and quantity for a SHORT position.
    SL is ABOVE entry, TP is BELOW entry.
    """
    if entry_price <= 0:
        raise ValueError("entry_price must be > 0")
    if entry_price < config.min_entry_price:
        raise ValueError(f"price {entry_price:.6f} below {config.min_entry_price}")

    capital = override_capital if override_capital is not None else config.capital_deployed
    if capital <= 0:
        raise ValueError("capital must be > 0")

    max_notional = capital * config.leverage
    sl_loss = capital * config.sl_capital_pct

    if atr_value and atr_value > 0:
        sl_dist = atr_value * config.atr_sl_multiplier
        raw_qty = min(sl_loss / sl_dist, max_notional / entry_price)
    else:
        raw_qty = max_notional / entry_price
        sl_dist = None

    qty = min(math.floor(raw_qty), config.max_quantity)
    if qty < 1:
        raise ValueError("quantity < 1")
    if qty * entry_price > max_notional * 1.01:
        qty = math.floor(max_notional / entry_price)
        if qty < 1:
            raise ValueError("quantity < 1 after cap")

    # SHORT: SL above entry, TP below entry
    if sl_dist is not None:
        sl_price = entry_price + sl_dist
        tp_price = entry_price - sl_dist * config.tp_rr_ratio
    else:
        sl_price = entry_price + (sl_loss / qty)
        tp_price = entry_price - (sl_loss / qty) * config.tp_rr_ratio

    return float(sl_price), float(max(tp_price, 0.0)), int(qty)


def cap_pnl(net_pnl: float, trade_capital: float, config: FastSlowMAConfig) -> float:
    """Cap PnL to prevent exponential equity explosion [7]."""
    return max(min(net_pnl, trade_capital * config.max_pnl_impact_pct), -trade_capital)


# ═══════════════════════════════════════════════════════════════════════
# SIGNAL GENERATION
#
# Matches PineScript logic:
#   if ta.crossover(maMomentum, 0)  → close short, entry long
#   if ta.crossunder(maMomentum, 0) → close long, entry short (if enabled)
#
# The PineScript uses position flipping (close + reverse on crossover).
# In our implementation, each signal is independent — the backtest
# handles position management.
# ═══════════════════════════════════════════════════════════════════════

def prepare_indicators(df: pd.DataFrame, config: FastSlowMAConfig) -> pd.DataFrame:
    """Compute fast MA, slow MA, ATR, and crossover signals."""
    if df is None or df.empty:
        return pd.DataFrame()

    min_bars = max(config.fast_length, config.slow_length) + 5
    if len(df) < min_bars:
        return pd.DataFrame()

    prepared = df.copy().sort_values("time").reset_index(drop=True)

    # Compute MAs using the selected type
    prepared["fast_ma"] = get_ma(prepared, config.ma_type, config.fast_length)
    prepared["slow_ma"] = get_ma(prepared, config.ma_type, config.slow_length)

    # ATR for SL sizing
    prepared["atr14"] = calculate_atr(prepared, 14)

    # MA momentum (matches PineScript: fastMA >= slowMA ? 1 : -1)
    prepared["ma_momentum"] = 0
    prepared.loc[prepared["fast_ma"] >= prepared["slow_ma"], "ma_momentum"] = 1
    prepared.loc[prepared["fast_ma"] < prepared["slow_ma"], "ma_momentum"] = -1

    # MA separation as percentage of slow MA
    slow_safe = prepared["slow_ma"].replace(0, float("nan"))
    prepared["ma_separation_pct"] = (
        (prepared["fast_ma"] - prepared["slow_ma"]).abs() / slow_safe * 100
    )

    # Crossovers (matches PineScript ta.crossover/ta.crossunder)
    prepared["long_signal"] = detect_crossover(prepared["fast_ma"], prepared["slow_ma"])
    prepared["short_signal"] = detect_crossunder(prepared["fast_ma"], prepared["slow_ma"])

    # Trend state for logging
    prepared["trend"] = "neutral"
    prepared.loc[prepared["fast_ma"] > prepared["slow_ma"], "trend"] = "bullish"
    prepared.loc[prepared["fast_ma"] < prepared["slow_ma"], "trend"] = "bearish"

    return prepared


def get_long_signal(
    df: pd.DataFrame,
    config: FastSlowMAConfig = FAST_SLOW_DEFAULT_CONFIG,
    override_capital: float | None = None,
) -> Optional[dict]:
    """
    Generate a LONG signal when fast MA crosses above slow MA.
    Used for live trading — checks the most recent candle.
    """
    if not config.allow_long:
        return None

    prepared = prepare_indicators(df, config)
    if prepared.empty:
        return None

    last_idx = len(prepared) - 1
    last = prepared.iloc[last_idx]

    if not last["long_signal"]:
        return None

    entry_price = float(last["close"])
    atr_value = float(last["atr14"])
    fast_ma = float(last["fast_ma"])
    slow_ma = float(last["slow_ma"])

    try:
        sl, tp, qty = calculate_long_sl_tp(
            entry_price, config, override_capital, atr_value
        )
    except ValueError:
        return None

    risk = entry_price - sl
    rr = (tp - entry_price) / risk if risk > 0 else None

    return {
        "signal": "LONG",
        "strategy": "fast_slow_ma_long",
        "entry": entry_price,
        "stop_loss": sl,
        "take_profit": tp,
        "quantity": qty,
        "capital_deployed": override_capital or config.capital_deployed,
        "leverage": config.leverage,
        "risk_reward": float(rr) if rr else None,
        "atr14": atr_value,
        "fast_ma": fast_ma,
        "slow_ma": slow_ma,
        "ma_type": config.ma_type,
        "fast_length": config.fast_length,
        "slow_length": config.slow_length,
        "ma_separation_pct": float(last["ma_separation_pct"]),
        "trend": last["trend"],
        "entry_candle_time": last["time"],
    }


def get_short_signal(
    df: pd.DataFrame,
    config: FastSlowMAConfig = FAST_SLOW_DEFAULT_CONFIG,
    override_capital: float | None = None,
) -> Optional[dict]:
    """
    Generate a SHORT signal when fast MA crosses below slow MA.
    Used for live trading — checks the most recent candle.
    Only fires if config.allow_short is True.
    """
    if not config.allow_short:
        return None

    prepared = prepare_indicators(df, config)
    if prepared.empty:
        return None

    last_idx = len(prepared) - 1
    last = prepared.iloc[last_idx]

    if not last["short_signal"]:
        return None

    entry_price = float(last["close"])
    atr_value = float(last["atr14"])
    fast_ma = float(last["fast_ma"])
    slow_ma = float(last["slow_ma"])

    try:
        sl, tp, qty = calculate_short_sl_tp(
            entry_price, config, override_capital, atr_value
        )
    except ValueError:
        return None

    risk = sl - entry_price
    rr = (entry_price - tp) / risk if risk > 0 else None

    return {
        "signal": "SHORT",
        "strategy": "fast_slow_ma_short",
        "entry": entry_price,
        "stop_loss": sl,
        "take_profit": tp,
        "quantity": qty,
        "capital_deployed": override_capital or config.capital_deployed,
        "leverage": config.leverage,
        "risk_reward": float(rr) if rr else None,
        "atr14": atr_value,
        "fast_ma": fast_ma,
        "slow_ma": slow_ma,
        "ma_type": config.ma_type,
        "fast_length": config.fast_length,
        "slow_length": config.slow_length,
        "ma_separation_pct": float(last["ma_separation_pct"]),
        "trend": last["trend"],
        "entry_candle_time": last["time"],
    }


def get_exit_signal(
    df: pd.DataFrame,
    current_position: str,
    config: FastSlowMAConfig = FAST_SLOW_DEFAULT_CONFIG,
) -> Optional[str]:
    """
    Check if current position should be closed.

    PineScript logic:
    - If long and bearish crossunder → close long
    - If short and bullish crossover → close short

    Returns "CLOSE_LONG", "CLOSE_SHORT", or None.
    """
    prepared = prepare_indicators(df, config)
    if prepared.empty:
        return None

    last = prepared.iloc[-1]

    if current_position == "LONG" and last["short_signal"]:
        return "CLOSE_LONG"
    elif current_position == "SHORT" and last["long_signal"]:
        return "CLOSE_SHORT"

    return None

# BACKTEST-COMPATIBLE ENTRY FILTER
#
# This function is designed to be called from backtest.py [5]
# in the same pattern as mhr_basic_entry_filters in strategy_short.py [4]
# ═══════════════════════════════════════════════════════════════════════

def fast_slow_ma_entry_filter(
    prepared: pd.DataFrame,
    entry_idx: int,
    config: FastSlowMAConfig = FAST_SLOW_DEFAULT_CONFIG,
    direction: str = "LONG",
) -> Optional[dict]:
    """
    Entry filter compatible with backtest_symbol() loop.

    direction: "LONG" or "SHORT"
    Returns filter_info dict or None if no signal.
    """
    if entry_idx < max(config.fast_length, config.slow_length) + 2:
        return None

    entry = prepared.iloc[entry_idx]

    if direction == "LONG" and not entry.get("long_signal", False):
        return None
    elif direction == "SHORT" and not entry.get("short_signal", False):
        return None

    if direction == "LONG" and not config.allow_long:
        return None
    if direction == "SHORT" and not config.allow_short:
        return None

    fast_ma = float(entry["fast_ma"]) if "fast_ma" in prepared.columns else None
    slow_ma = float(entry["slow_ma"]) if "slow_ma" in prepared.columns else None
    ma_sep = float(entry["ma_separation_pct"]) if "ma_separation_pct" in prepared.columns else None

    return {
        "entry_candle_time": entry["time"],
        "fast_ma": fast_ma,
        "slow_ma": slow_ma,
        "ma_type": config.ma_type,
        "fast_length": config.fast_length,
        "slow_length": config.slow_length,
        "ma_separation_pct": ma_sep,
        "trend": entry.get("trend", "unknown"),
        "ema_distance_pct": ma_sep,    # compatibility with backtest logging
        "rsi14": None,
        "wick_body_ratio": 0.0,
    }


# ═══════════════════════════════════════════════════════════════════════
# BACKTEST EXIT SIMULATOR
#
# The PineScript uses position flipping — close + reverse on crossover.
# This simulator handles:
# 1. Hold until opposite crossover (close position)
# 2. OR SL/TP hit
# 3. OR max hold reached
# ═══════════════════════════════════════════════════════════════════════

def simulate_fast_slow_exit(
    candles: pd.DataFrame,
    entry_idx: int,
    entry_price: float,
    sl_price: float,
    tp_price: float,
    qty: int,
    direction: str,
    config: FastSlowMAConfig,
    max_hold: int = 100,
) -> tuple[float, str, pd.Timestamp, float, int]:
    """
    Simulate exit for fast/slow MA strategy.

    The PineScript exits on the OPPOSITE crossover signal.
    We also check SL/TP levels per candle.

    Returns: (pnl, exit_reason, exit_time, exit_price, bars_held)
    """
    bars_held = 0
    max_exit_idx = min(len(candles) - 1, entry_idx + max_hold)

    for idx in range(entry_idx + 1, max_exit_idx + 1):
        bars_held += 1
        candle = candles.iloc[idx]
        c_high = float(candle["high"])
        c_low = float(candle["low"])
        c_close = float(candle["close"])

        if direction == "LONG":
            # Check SL (price drops below SL)
            if c_low <= sl_price:
                pnl = (sl_price - entry_price) * qty
                return pnl, "SL_HIT", candle["time"], sl_price, bars_held

            # Check TP (price rises above TP)
            if c_high >= tp_price:
                pnl = (tp_price - entry_price) * qty
                return pnl, "TP_HIT", candle["time"], tp_price, bars_held

            # Check for exit signal (bearish crossover)
            if "short_signal" in candles.columns and candle.get("short_signal", False):
                pnl = (c_close - entry_price) * qty
                return pnl, "MA_CROSS_EXIT", candle["time"], c_close, bars_held

        else:  # SHORT
            # Check SL (price rises above SL)
            if c_high >= sl_price:
                pnl = (entry_price - sl_price) * qty
                return pnl, "SL_HIT", candle["time"], sl_price, bars_held

            # Check TP (price drops below TP)
            if c_low <= tp_price:
                pnl = (entry_price - tp_price) * qty
                return pnl, "TP_HIT", candle["time"], tp_price, bars_held

            # Check for exit signal (bullish crossover)
            if "long_signal" in candles.columns and candle.get("long_signal", False):
                pnl = (entry_price - c_close) * qty
                return pnl, "MA_CROSS_EXIT", candle["time"], c_close, bars_held

    # Max hold exit
    last = candles.iloc[max_exit_idx]
    exit_price = float(last["close"])
    if direction == "LONG":
        pnl = (exit_price - entry_price) * qty
    else:
        pnl = (entry_price - exit_price) * qty

    return pnl, "MAX_HOLD_EXIT", last["time"], exit_price, bars_held


# ═══════════════════════════════════════════════════════════════════════
# VOLUME HELPER (kept for backtest compatibility)
# ═══════════════════════════════════════════════════════════════════════

def estimate_candle_volume_usd(candles: pd.DataFrame, lookback_bars: int = 24) -> float:
    if candles.empty or "volume" not in candles.columns:
        return 0.0
    recent = candles.tail(lookback_bars)
    if "close" in candles.columns:
        return float((recent["volume"] * recent["close"]).sum())
    return float(recent["volume"].sum())
